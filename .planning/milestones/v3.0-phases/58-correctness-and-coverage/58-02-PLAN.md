import { convexTest } from "convex-test";
import { describe, it, expect, vi, beforeEach } from "vitest";
import { internal } from "../_generated/api";
import schema from "../schema";

// Set required env vars before module load (assertPlanPriceConfig reads these)
process.env.STRIPE_SECRET_KEY = "sk_test_fake";
process.env.STRIPE_GO_PRICE_ID = "price_go_test";
process.env.STRIPE_PRO_PRICE_ID = "price_pro_test";

// Hoist mock functions so they're available in vi.mock factory
const { mockRetrieve } = vi.hoisted(() => ({
  mockRetrieve: vi.fn(),
}));

vi.mock("stripe", () => {
  return {
    default: class MockStripe {
      subscriptions = { retrieve: mockRetrieve };
    },
  };
});

// Helper: seed a user + subscription record in the DB
async function seedSubscription(
  t: ReturnType<typeof convexTest>,
  overrides: {
    status?: string;
    plan?: string;
    stripeSubscriptionId?: string;
    currentPeriodEnd?: number;
    cancelAtPeriodEnd?: boolean;
  } = {}
) {
  return await t.run(async (ctx) => {
    const now = Date.now();
    const userId = await ctx.db.insert("users", {
      clerkId: "clerk_sub_test",
      email: "sub@test.com",
      firstName: "Sub",
      lastName: "Tester",
      state: "active" as const,
      createdAt: now,
      updatedAt: now,
      lastLoginAt: now,
    });

    const subscriptionId = await ctx.db.insert("subscriptions", {
      userId,
      stripeCustomerId: "cus_existing",
      stripeSubscriptionId: overrides.stripeSubscriptionId ?? "sub_123",
      status: (overrides.status ?? "active") as "active",
      plan: (overrides.plan ?? "go") as "go",
      currentPeriodStart: now - 30 * 24 * 60 * 60 * 1000,
      currentPeriodEnd: overrides.currentPeriodEnd ?? now + 30 * 24 * 60 * 60 * 1000,
      cancelAtPeriodEnd: overrides.cancelAtPeriodEnd ?? false,
      createdAt: now,
      updatedAt: now,
    });

    return { userId, subscriptionId };
  });
}

describe("checkSubscriptionStatus", () => {
  beforeEach(() => {
    mockRetrieve.mockReset();
  });

  it("corrects divergence with string customer ID", async () => {
    const t = convexTest(schema);
    const { subscriptionId } = await seedSubscription(t, {

    expect(sub!.plan).toBe("pro");
  });

      stripeSubscriptionId: "sub_exp_test",
    });

    // Stripe returns diverged status with EXPANDED customer object
    mockRetrieve.mockResolvedValue({

**Why tests/ not convex/:** The `vitest.config.ts` maps `convex/**/*.test.ts` to `edge-runtime`. The `grobid.ts` module imports `cheerio` which requires Node APIs. Placing the test in `tests/` ensures it runs with `environment: "node"` where cheerio works.

```typescript
import { describe, it, expect, vi, beforeEach, afterEach } from "vitest";

// Mock cheerio before importing grobid to avoid XML parsing in tests
vi.mock("cheerio", () => ({
  load: vi.fn(() => {
    const mockQuery = (selector: string) => ({
      first: () => ({ text: () => "Mock Title" }),
    });
    mockQuery.xml = true;
    return mockQuery;
  }),
}));

describe("GROBID URL discovery fallback", () => {
  const originalGrobidUrl = process.env.GROBID_URL;
  let mockFetch: ReturnType<typeof vi.fn>;

  beforeEach(() => {
    mockFetch = vi.fn();
    vi.stubGlobal("fetch", mockFetch);
    // Clear GROBID_URL to test default fallback chain
    delete process.env.GROBID_URL;
    vi.resetModules();
  });

  afterEach(() => {
    vi.unstubAllGlobals();
    if (originalGrobidUrl === undefined) {
      delete process.env.GROBID_URL;
    } else {
      process.env.GROBID_URL = originalGrobidUrl;
    }
    vi.resetModules();
  });

  it("throws descriptive error when all endpoints fail", async () => {
    // Re-import after env var reset to get fresh module-level constants
    const { processFulltextDocumentWithGrobid } = await import(
      "@/convex/extraction/grobid"
    );

    // All endpoints fail with connection refused
    mockFetch.mockRejectedValue(new Error("Connection refused"));

    const pdfBuffer = new ArrayBuffer(8);

    await expect(
      processFulltextDocumentWithGrobid(pdfBuffer)
    ).rejects.toThrow("GROBID parsing failed across");

    // Verify multiple URLs were attempted (at least the default fallback chain)
    expect(mockFetch).toHaveBeenCalledTimes(3); // Docker, localhost, host.docker.internal
    // Note: localhost and DEFAULT_GROBID_URL deduplicate (both http://localhost:8070)
  });

  it("succeeds when first endpoint fails but second succeeds", async () => {
    const { processFulltextDocumentWithGrobid } = await import(
      "@/convex/extraction/grobid"
    );

    const validTeiXml = `<?xml version="1.0" encoding="UTF-8"?>
<TEI xmlns="http://www.tei-c.org/ns/1.0">
  <teiHeader>
    <fileDesc>
      <titleStmt><title>Test Paper</title></titleStmt>
    </fileDesc>
  </teiHeader>
  <text>
    <body>
      <div>
        <head>Introduction</head>
        <p>This is a test paragraph with some content.</p>
      </div>
    </body>
  </text>
</TEI>`;

    // First call fails, second call succeeds
    mockFetch
      .mockRejectedValueOnce(new Error("Connection refused"))
      .mockResolvedValueOnce({
        ok: true,
        text: async () => validTeiXml,
      });

    const pdfBuffer = new ArrayBuffer(8);
    const result = await processFulltextDocumentWithGrobid(pdfBuffer);

    // Function should have tried 2 endpoints
    expect(mockFetch).toHaveBeenCalledTimes(2);

    // Should return a valid result from the second endpoint
    expect(result).toBeDefined();
    expect(result.teiXml).toContain("<TEI");
    expect(result.title).toBe("Test Paper");
    expect(result.blocks.length).toBeGreaterThan(0);
  });

  it("uses explicit baseUrl first in the fallback chain", async () => {
    const { processFulltextDocumentWithGrobid } = await import(
      "@/convex/extraction/grobid"
    );

    // All calls fail -- we just want to verify the order of URLs attempted
    mockFetch.mockRejectedValue(new Error("Connection refused"));

    const pdfBuffer = new ArrayBuffer(8);
    const explicitUrl = "http://custom-grobid:9999";

    await expect(
      processFulltextDocumentWithGrobid(pdfBuffer, { baseUrl: explicitUrl })
    ).rejects.toThrow("GROBID parsing failed across");

    // First call should be to the explicit URL
    const firstCallUrl = mockFetch.mock.calls[0][0];
    expect(firstCallUrl).toContain("custom-grobid:9999");
  });
});
```

**Key design choices:**
- Placed in `tests/` (not `convex/`) to run in node environment where cheerio works
- `vi.stubGlobal("fetch")` replaces global fetch without requiring msw or nock
- `vi.resetModules()` ensures fresh module-level constant evaluation after env var changes
- Dynamic `await import()` after `vi.resetModules()` to re-evaluate `DEFAULT_GROBID_URL`
- Mock cheerio to avoid needing real XML parsing in fallback-chain tests (the second test case uses real TEI XML that cheerio can parse, so we may need to conditionally use the real cheerio -- adjust based on runtime behavior)
- 3 test cases: all-fail, first-fail-second-succeed, explicit URL priority
