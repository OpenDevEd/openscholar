export const FREE_SIGNUP_CREDIT_GRANT = 2_000;

export const PLAN_CONFIG = {
  free: { creditsPerMonth: 0, tier: 0, freeCreditCap: FREE_SIGNUP_CREDIT_GRANT },
  go: { creditsPerMonth: 3_000, tier: 10, freeCreditCap: 6_000 },
  pro: { creditsPerMonth: 15_000, tier: 20, freeCreditCap: 30_000 },
  enterprise: { creditsPerMonth: 150_000, tier: 30, freeCreditCap: 300_000 },
} as const;

// Price IDs from environment - evaluated at module load time
export const GO_PRICE_ID = process.env.STRIPE_GO_PRICE_ID || "";
export const PRO_PRICE_ID = process.env.STRIPE_PRO_PRICE_ID || "";
export const ENTERPRISE_PRICE_ID = process.env.STRIPE_ENTERPRISE_PRICE_ID || "";

export function getPlanFromPriceId(priceId?: string): PlanType {
  if (!priceId) return "free";
  if (priceId === GO_PRICE_ID) return "go";
  if (priceId === PRO_PRICE_ID) return "pro";
  if (ENTERPRISE_PRICE_ID && priceId === ENTERPRISE_PRICE_ID) return "enterprise";
  // Fallback: name-based matching
  if (priceId.includes("enterprise")) return "enterprise";
  if (priceId.includes("pro")) return "pro";
  return "free";
}

export function getPlanTier(plan: PlanType): number {
  return PLAN_CONFIG[plan]?.tier ?? 0;
}

export function getFreeCreditCapForPlan(plan: PlanType): number {
  return PLAN_CONFIG[plan]?.freeCreditCap ?? PLAN_CONFIG.free.freeCreditCap;
}
```

export function requireCreditsEnabled(): void {
  if (process.env.NEXT_PUBLIC_CREDITS_ENABLED === "false") {
    throw new Error("Credits feature is disabled");
  }
}

export function requireCreditTopupsEnabled(): void {
  if (process.env.NEXT_PUBLIC_CREDIT_TOPUPS_ENABLED === "false") {
    throw new Error("Credit top-ups feature is disabled");
  }
}

export function requireSubscriptionsEnabled(): void {
  if (process.env.NEXT_PUBLIC_SUBSCRIPTIONS_ENABLED === "false") {
    throw new Error("Subscriptions feature is disabled");
  }
}
```

Note: Feature flag functions check process.env at CALL TIME (not module load). However, success criteria requires vi.resetModules() for env var isolation, so use dynamic import pattern.

Note: getPlanFromPriceId compares against GO_PRICE_ID/PRO_PRICE_ID/ENTERPRISE_PRICE_ID which are set at MODULE LOAD TIME. To test the env-var-based matching, use vi.resetModules() + dynamic import with env vars set before import. For fallback name-based matching (priceId.includes("enterprise")/("pro")), no env setup is needed.
