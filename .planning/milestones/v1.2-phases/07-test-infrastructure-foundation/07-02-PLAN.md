```typescript
import Stripe from "stripe";

/**
 * Create a Stripe client for E2E billing tests.
 * Uses STRIPE_SECRET_KEY from environment.
 */
export function createStripeClient(): Stripe {
  const secretKey = process.env.STRIPE_SECRET_KEY;
  if (!secretKey) {
    throw new Error("Missing STRIPE_SECRET_KEY for billing E2E tests");
  }
  return new Stripe(secretKey, {
    apiVersion: "2026-01-28.clover",
  });
}

/**
 * Attach a test Visa payment method to a Stripe customer and set it as default.
 * Returns the payment method ID.
 */
export async function attachTestPaymentMethod(args: {
  stripe: Stripe;
  customerId: string;
}): Promise<string> {
  const pm = await args.stripe.paymentMethods.create({
    type: "card",
    card: { token: "tok_visa" },
  });
  await args.stripe.paymentMethods.attach(pm.id, {
    customer: args.customerId,
  });
  await args.stripe.customers.update(args.customerId, {
    invoice_settings: { default_payment_method: pm.id },
  });
  return pm.id;
}

/**
 * Create a Stripe subscription via checkout-bypass pattern.
 * Creates a customer, attaches a test payment method, and creates a subscription.
 * This avoids automating the Stripe Checkout hosted page (which Stripe blocks).
 *
 * The `userId` metadata is critical -- Convex webhook handlers use it to link
 * the subscription to the correct Convex user.
 */
export async function createSubscriptionBypass(args: {
  stripe: Stripe;
  email: string;
  userId: string;
  priceId: string;
}): Promise<{
  customerId: string;
  subscriptionId: string;
  invoiceId: string;
  paymentMethodId: string;
}> {
  // 1. Create Stripe customer with userId metadata
  const customer = await args.stripe.customers.create({
    email: args.email,
    metadata: { userId: args.userId },
  });

  // 2. Attach test payment method
  const paymentMethodId = await attachTestPaymentMethod({
    stripe: args.stripe,
    customerId: customer.id,
  });

  // 3. Create subscription (triggers webhooks to Convex)
  const subscription = await args.stripe.subscriptions.create({
    customer: customer.id,
    items: [{ price: args.priceId }],
    payment_behavior: "error_if_incomplete",
    metadata: { userId: args.userId },
    default_payment_method: paymentMethodId,
    expand: ["latest_invoice"],
  });

  const invoiceId =
    typeof subscription.latest_invoice === "string"
      ? subscription.latest_invoice
      : subscription.latest_invoice?.id;

  if (!invoiceId) {
    throw new Error(
      `Subscription ${subscription.id} missing latest invoice ID`,
    );
  }

  return {
    customerId: customer.id,
    subscriptionId: subscription.id,
    invoiceId,
    paymentMethodId,
  };
}

/**
 * Clean up all Stripe objects for a test customer.
 * Cancels active subscriptions and deletes the customer.
 */
export async function cleanupStripeCustomer(args: {
  stripe: Stripe;
  customerId: string;
}): Promise<void> {
  // Cancel all active subscriptions
  const subscriptions = await args.stripe.subscriptions.list({
    customer: args.customerId,
    status: "all",
    limit: 100,
  });

  for (const sub of subscriptions.data) {
    if (sub.status !== "canceled") {
      try {
        await args.stripe.subscriptions.cancel(sub.id);
      } catch {
        // Best-effort cleanup -- subscription may already be canceled
      }
    }
  }

  // Delete the customer
  try {
    await args.stripe.customers.del(args.customerId);
  } catch {
    // Best-effort cleanup -- customer may already be deleted
  }
}
```

Create `e2e/billing/webhook-waiter.ts` -- a utility that polls Convex for billing state convergence before E2E tests make UI assertions.

This adapts the `waitForCondition` + `runConvexFunction` + `getBillingSnapshotForUser` patterns from `tests/live-billing/support.ts` into a class-based API optimized for E2E test readability.

**Implementation:**

```typescript
import { execFile } from "node:child_process";
import { promisify } from "node:util";

const execFileAsync = promisify(execFile);

/**
 * Snapshot of billing state from Convex, matching the shape returned by
 * testing/billing:getBillingSnapshotForUser internal query.
 */
export type BillingSnapshot = {
  user: {
    _id: string;
    clerkId: string;
    email: string;
    state: string;
  } | null;
  credits: Array<{
    freeBalance: number;
    paidBalance: number;
    lifetimeUsed: number;
  }>;
  subscriptions: Array<{
    plan: string;
    status: string;
    stripeSubscriptionId?: string;
    stripeCustomerId?: string;
    creditsPerMonth?: number;
  }>;
  billingCustomers: Array<{
    stripeCustomerId: string;
  }>;
};

/**
 * Polls Convex backend for billing state convergence.
 *
 * After Stripe API calls (subscription creation, payment, plan change),
 * webhooks are processed asynchronously by Convex. This utility blocks
 * until the expected state is reflected in the Convex database before
 * the test proceeds to make UI assertions.
 *
 * Uses `npx convex run` CLI to query the internal testing function.
 *
 * Usage:
 * ```typescript
 * const waiter = new WebhookWaiter(convexUserId);
 * await waiter.waitForSubscriptionPlan("go");
 * await waiter.waitForCreditBalance((total) => total >= 3000);
 * ```
 */
export class WebhookWaiter {
  constructor(
    private convexUserId: string,
    private defaultTimeoutMs: number = 30_000,
    private intervalMs: number = 2_500,
  ) {}

  /**
   * Wait for the user's subscription to reach the expected plan and active status.
   */
  async waitForSubscriptionPlan(
    expectedPlan: string,
    timeoutMs?: number,
  ): Promise<void> {
    await this.waitForCondition(
      `subscription plan = "${expectedPlan}"`,
      async () => {
        const snapshot = await this.getBillingSnapshot();
        const sub = snapshot.subscriptions[0];
        return sub?.plan === expectedPlan && sub?.status === "active"
          ? true
          : null;
      },
      timeoutMs,
    );
  }

  /**
   * Wait for the user's total credit balance to satisfy a predicate.
   */
  async waitForCreditBalance(
    predicate: (totalBalance: number) => boolean,
    timeoutMs?: number,
  ): Promise<number> {
    return await this.waitForCondition(
      "credit balance matches predicate",
      async () => {
        const snapshot = await this.getBillingSnapshot();
        const credits = snapshot.credits[0];
        if (!credits) return null;
        const total = credits.freeBalance + credits.paidBalance;
        return predicate(total) ? total : null;
      },
      timeoutMs,
    );
  }

  /**
   * Wait for a billing customer link to exist for the user.
   */
  async waitForBillingCustomerLink(timeoutMs?: number): Promise<string> {
    return await this.waitForCondition(
      "billing customer link exists",
      async () => {
        const snapshot = await this.getBillingSnapshot();
        const customer = snapshot.billingCustomers[0];
        return customer?.stripeCustomerId ?? null;
      },
      timeoutMs,
    );
  }

  /**
   * Get the current billing snapshot without waiting.
   */
  async getBillingSnapshot(): Promise<BillingSnapshot> {
    return await this.runConvexQuery(
      "testing/billing:getBillingSnapshotForUser",
      { userId: this.convexUserId },
    );
  }

  /**
   * Reset all billing state for the user in Convex.
   * Called during fixture teardown.
   */
  async resetBillingState(): Promise<void> {
    await this.runConvexQuery(
      "testing/billing:resetBillingStateForUser",
      { userId: this.convexUserId },
    );
  }

  /**
   * Poll a condition until it returns a non-null value or times out.
   * Adapted from tests/live-billing/support.ts waitForCondition.
   */
  private async waitForCondition<T>(
    description: string,
    check: () => Promise<T | null>,
    timeoutMs?: number,
  ): Promise<T> {
    const timeout = timeoutMs ?? this.defaultTimeoutMs;
    const startedAt = Date.now();
    let lastError: string | null = null;

    while (Date.now() - startedAt <= timeout) {
      try {
        const value = await check();
        if (value !== null && value !== undefined) {
          return value;
        }
      } catch (error) {
        lastError = error instanceof Error ? error.message : String(error);
      }
      await new Promise((r) => setTimeout(r, this.intervalMs));
    }

    const suffix = lastError ? ` Last error: ${lastError}` : "";
    throw new Error(
      `WebhookWaiter: timed out waiting for ${description} after ${timeout}ms.${suffix}`,
    );
  }

  /**
   * Run a Convex internal function via CLI.
   * Adapted from tests/live-billing/support.ts runConvexFunction.
   */
  private async runConvexQuery<T>(
    functionPath: string,
    payload: Record<string, unknown>,
  ): Promise<T> {
    const cliArgs = [
      "convex",
      "run",
      functionPath,
      JSON.stringify(payload),
      "--typecheck",
      "disable",
      "--codegen",
      "disable",
    ];

    if (process.env.CONVEX_DEPLOYMENT_NAME) {
      cliArgs.push("--deployment-name", process.env.CONVEX_DEPLOYMENT_NAME);
    }

    const { stdout } = await execFileAsync("npx", cliArgs, {
      env: process.env,
      maxBuffer: 1024 * 1024 * 8,
      timeout: 15_000,
    });

    const trimmed = stdout.trim();
    if (!trimmed) {
      throw new Error(`Empty response from convex run ${functionPath}`);
    }

    // Parse stdout -- handle Convex CLI output format
    const lastLine = trimmed.split("\n").at(-1)?.trim() ?? "";
    for (const attempt of [trimmed, lastLine]) {
      try {
        return JSON.parse(attempt) as T;
      } catch {
        // Try next
      }
    }

    throw new Error(
      `Failed to parse convex run output for ${functionPath}: ${trimmed.slice(0, 200)}`,
    );
  }
}
```

Key design decisions:
- **CLI-based polling** (not DOM-based): Works regardless of current page, more reliable for checking exact backend state. The research recommends implementing both but starting with CLI as primary.
- **Class-based API**: Constructor takes `convexUserId` once; individual `waitFor*` methods are clean one-liners in tests.
- **Reuses `tests/live-billing/support.ts` patterns**: `waitForCondition` retry loop, `runConvexFunction` CLI invocation, `getBillingSnapshotForUser` query.
- **Includes `resetBillingState`**: Fixture teardown needs this to clean up Convex data.
- **30s default timeout, 2.5s poll interval**: Same as live-billing tests, proven to work for Convex webhook convergence.
