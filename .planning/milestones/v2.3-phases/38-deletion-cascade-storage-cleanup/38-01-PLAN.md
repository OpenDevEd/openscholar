  ```typescript
  export const cleanupFailedPaperStorage = internalMutation({
    args: { jobId: v.id("extractionJobs") },
    returns: v.object({ cleaned: v.number() }),
    handler: async (ctx, args): Promise<{ cleaned: number }> => {
      const papersWithStorage = await ctx.db
        .query("extractionPapers")
        .withIndex("by_job", (q) => q.eq("jobId", args.jobId))
        .filter((q) => q.neq(q.field("storageId"), undefined))
        .collect();

      let cleaned = 0;
      for (const paper of papersWithStorage) {
        if (!paper.storageId) continue;
        try {
          await ctx.storage.delete(paper.storageId);
          await ctx.db.patch(paper._id, { storageId: undefined });
          cleaned++;
        } catch (e) {
          console.error(`Failed to delete storage for paper ${paper._id}:`, e);
        }
      }

      return { cleaned };
    },
  });
  ```

  **Rules:**
  - Place this function between `deleteExtractionInternal` and the `// Internal mutations (called by workflow/actions)` comment block.
  - The mutation is `jobId`-scoped — it does NOT do a global cross-job scan. This avoids needing a new schema index.
  - `ctx.storage.delete` errors MUST be caught per paper — errors on one paper must not stop cleanup of subsequent papers.
  - After a successful `ctx.storage.delete`, ALWAYS patch `storageId: undefined` on the paper record so subsequent calls are idempotent.
  - The `.filter()` call uses `q.neq(q.field("storageId"), undefined)` to skip papers with no storageId. This is a post-index filter — it is correct for this use case (job-scoped, not a global scan).
  - Do NOT export this from `convex/extraction/index.ts` — it is `internal` only.
