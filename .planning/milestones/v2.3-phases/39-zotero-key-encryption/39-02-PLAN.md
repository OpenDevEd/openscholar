**Add `migrateToEncryptedKeys` to `convex/zotero/actions.ts`:**

Add an `internalAction` at the end of the file (before any closing braces or after the last export). It must use the `listAllConnections` query and `patchEncryptedKey` mutation created in Plan 01.

```typescript
export const migrateToEncryptedKeys = internalAction({
  args: {},
  returns: v.object({
    total: v.number(),
    migrated: v.number(),
    skipped: v.number(),
  }),
  handler: async (ctx) => {
    const encryptionKey = process.env.ENCRYPTION_KEY;
    if (!encryptionKey) {
      throw createError("API_KEY_ENCRYPTION_KEY_MISSING", "Encryption key not configured");
    }

    const connections = await ctx.runQuery(internal.zotero.internal.listAllConnections);
    let migrated = 0;
    let skipped = 0;

    for (const conn of connections) {
      if (conn.encryptedApiKey) {
        skipped++;
        continue;
      }

      const encrypted = encrypt(conn.apiKey, encryptionKey);
      const prefix = getKeyPrefix(conn.apiKey);
      await ctx.runMutation(internal.zotero.mutations.patchEncryptedKey, {
        connectionId: conn._id,
        encryptedApiKey: encrypted,
        keyPrefix: prefix,
      });
      migrated++;
    }

    return { total: connections.length, migrated, skipped };
  },
});
```

Make sure `internalAction` is imported from `"../_generated/server"` (it should already be from Plan 01 or existing imports). The `encrypt` and `getKeyPrefix` imports from `"../lib/crypto"` should already exist from Plan 01.

Update `resolveZoteroApiKey` to handle the optional apiKey gracefully:
```typescript
function resolveZoteroApiKey(connection: NonNullable<ZoteroConnection>): string {
  if (connection.encryptedApiKey) {
    const encryptionKey = process.env.ENCRYPTION_KEY;
    if (!encryptionKey) {
      throw createError("API_KEY_ENCRYPTION_KEY_MISSING", "Encryption key not configured");
    }
    return decrypt(connection.encryptedApiKey, encryptionKey);
  }
  if (connection.apiKey) {
    return connection.apiKey;
  }
  throw createError("ZOTERO_NOT_CONNECTED", "Zotero connection has no API key");
}
```
