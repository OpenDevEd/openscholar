import { describe, it, expect } from "vitest";
import enUS from "../messages/en-US.json";
import enGB from "../messages/en-GB.json";
import fr from "../messages/fr.json";
import ja from "../messages/ja.json";
import zh from "../messages/zh.json";
import hi from "../messages/hi.json";

// All new keys added for extraction insights (Phase 83+)
// Existing keys (confidence, avgConf, etc.) are already covered by component usage
const EXTRACTION_INSIGHTS_NEW_KEYS = [
  "confidenceAbstractOnly",
  "confidenceNoContent",
  "confidenceCapExplained",
  "barScalingWarning",
  "progressBarLabel",
  "statusDotComplete",
  "statusDotPartial",
  "statusDotEmpty",
  "answerGrouping",
  "answerGroupingExpand",
  "answerGroupingVariants",
] as const;

type LocaleMessages = typeof enUS;

const locales: Record<string, LocaleMessages> = {
  "en-US": enUS,
  "en-GB": enGB,
  fr: fr as unknown as LocaleMessages,
  ja: ja as unknown as LocaleMessages,
  zh: zh as unknown as LocaleMessages,
  hi: hi as unknown as LocaleMessages,
};

describe("extraction insights i18n coverage", () => {
  it("all new extraction keys exist in all 6 locales", () => {
    for (const [locale, messages] of Object.entries(locales)) {
      const extraction = messages.extraction as Record<string, unknown>;
      for (const key of EXTRACTION_INSIGHTS_NEW_KEYS) {
        expect(
          extraction,
          `Missing key "${key}" in locale "${locale}"`
        ).toHaveProperty(key);
      }
    }
  });

  it("no translation value is an empty string", () => {
    for (const [locale, messages] of Object.entries(locales)) {
      const extraction = messages.extraction as Record<string, unknown>;
      for (const key of EXTRACTION_INSIGHTS_NEW_KEYS) {
        expect(
          extraction[key],
          `Empty value for key "${key}" in locale "${locale}"`
        ).not.toBe("");
      }
    }
  });

  it("non-English locales do not contain untranslated key names as values", () => {
    for (const [locale, messages] of Object.entries(locales)) {
      if (locale === "en-US" || locale === "en-GB") continue;
      const extraction = messages.extraction as Record<string, unknown>;
      for (const key of EXTRACTION_INSIGHTS_NEW_KEYS) {
        // The value should not equal the key itself (copy-paste error)
        expect(
          extraction[key],
          `Value equals key name "${key}" in locale "${locale}" — likely untranslated`
        ).not.toBe(key);
      }
    }
  });
});

The goal is to test the component's rendered DOM structure and behavior, NOT implementation details — so assertions use `data-testid` attributes and semantic queries.

1. Mock the query to return a known insights payload

import { describe, it, expect, vi } from "vitest";
import { render, screen } from "@testing-library/react";
import { InsightsOverview } from "../components/extraction/insights-overview";
import type { Id } from "../convex/_generated/dataModel";

// Mock next-intl — return key as value (avoids NextIntlClientProvider setup)

const mockInsights = {
  status: "ok" as const,
  summary: {
    answeredRate: 0.75,
    groundedRate: 0.6,
    bestEffortRate: 0.15,
    noContentCount: 3,
  },
  questions: [
    { key: "sample_size", question: "What is the sample size?", answered: 7, total: 10, grounded: 5, avgConfidence: 0.82 },
    { key: "methods", question: "What methods were used?", answered: 9, total: 10, grounded: 8, avgConfidence: 0.91 },
  ],
  diagnostics: { noContentPapers: [] },
};

vi.mock("@/lib/auth/use-auth-query", () => ({
  useProtectedQuery: () => ({ data: mockInsights, isDeferred: false }),
}));

const mockJobId = "mock-job-id" as Id<"extractionJobs">;

describe("InsightsOverview component", () => {
  it("renders null when enabled=false", () => {
    const { container } = render(
      <InsightsOverview
        jobId={mockJobId}
        onSelectQuestion={() => {}}
        enabled={false}
      />
    );
    expect(container.firstChild).toBeNull();
  });

  it("renders 4 summary metric sections", () => {
    const { container } = render(
      <InsightsOverview
        jobId={mockJobId}
        onSelectQuestion={() => {}}
        enabled={true}
      />
    );
    // Grid of cards: answered, grounded, bestEffort, noContent
    // Look for percentage text matching the mock data
    expect(container.textContent).toContain("75%"); // answeredRate
    expect(container.textContent).toContain("60%"); // groundedRate

    expect(container.textContent).toContain("3");   // noContentCount
  });

  it("renders question list with one button per question", () => {
    render(
      <InsightsOverview
        jobId={mockJobId}
        onSelectQuestion={() => {}}
        enabled={true}
      />
    );
    // Each question is rendered as a <button>
    const buttons = screen.getAllByRole("button");
    // At least 2 question buttons (may be more if rerun button present)

  it("calls onSelectQuestion with the question key when a question button is clicked", async () => {

  });
});
