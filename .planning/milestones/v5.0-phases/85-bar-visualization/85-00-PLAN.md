    ```typescript
    import { describe, it, expect } from "vitest";
    import { normalizeToMax, scaleAnswersForDisplay } from "../lib/extraction/bar-scaling";

    describe("normalizeToMax", () => {
      it("maps dominant value to 1.0 and scales others proportionally", () => {
        const result = normalizeToMax([85, 10, 5]);
        expect(result[0]).toBe(1.0);
        expect(result[1]).toBeCloseTo(10 / 85);
        expect(result[2]).toBeCloseTo(5 / 85);
      });

      it("returns all-zeros for all-zero input without NaN", () => {
        const result = normalizeToMax([0, 0, 0]);
        expect(result).toEqual([0, 0, 0]);
        result.forEach(v => expect(Number.isNaN(v)).toBe(false));
      });

      it("returns [1.0] for single-item input", () => {
        expect(normalizeToMax([42])).toEqual([1.0]);
      });

      it("treats NaN as 0 and never outputs NaN", () => {
        const result = normalizeToMax([NaN, 5]);
        expect(result[0]).toBe(0);
        expect(result[1]).toBe(1.0);
      });

      it("returns empty array for empty input", () => {
        expect(normalizeToMax([])).toEqual([]);
      });
    });

    describe("scaleAnswersForDisplay", () => {
      it("dominant answer gets scaledWidth=1.0, minority answer proportional", () => {
        const result = scaleAnswersForDisplay([
          { answer: "Yes", count: 80 },
          { answer: "No", count: 20 },
        ]);
        expect(result[0].scaledWidth).toBe(1.0);
        expect(result[0].percentage).toBe(80);
        expect(result[1].scaledWidth).toBe(0.25);
        expect(result[1].percentage).toBe(20);
      });

      it("all-zero counts produce all scaledWidth=0 without NaN", () => {
        const result = scaleAnswersForDisplay([
          { answer: "Yes", count: 0 },
          { answer: "No", count: 0 },
        ]);
        result.forEach(r => {
          expect(r.scaledWidth).toBe(0);
          expect(Number.isNaN(r.percentage)).toBe(false);
        });
      });

      it("single answer gets scaledWidth=1.0, percentage=100", () => {
        const result = scaleAnswersForDisplay([{ answer: "Yes", count: 50 }]);
        expect(result[0].scaledWidth).toBe(1.0);
        expect(result[0].percentage).toBe(100);
      });
    });
    ```

    ```typescript
    import { describe, it, expect } from "vitest";
    import { validateBarData } from "../lib/extraction/bar-validation";
    import type { SafeBarData } from "../lib/extraction/bar-validation";

    describe("validateBarData", () => {
      it("returns isEmpty=true for null input", () => {
        const result = validateBarData(null as any);
        expect(result.isEmpty).toBe(true);
        expect(result.max).toBe(0);
        expect(result.items).toEqual([]);
      });

      it("returns isEmpty=true for empty array", () => {
        const result = validateBarData([]);
        expect(result.isEmpty).toBe(true);
      });

      it("returns isEmpty=true when all counts are zero", () => {
        const result = validateBarData([
          { answer: "A", count: 0 },
          { answer: "B", count: 0 },
        ]);
        expect(result.isEmpty).toBe(true);
      });

      it("filters NaN/negative counts; isEmpty=true if nothing left", () => {
        const result = validateBarData([
          { answer: "A", count: NaN },
          { answer: "B", count: -5 },
        ]);
        expect(result.isEmpty).toBe(true);
      });

      it("returns valid items with percentage for normal input", () => {
        const result = validateBarData([
          { answer: "A", count: 90 },
          { answer: "B", count: 10 },
        ]);
        expect(result.isEmpty).toBe(false);
        expect(result.max).toBe(90);
        expect(result.items).toHaveLength(2);
        expect(result.items[0].percentage).toBe(90);
        expect(result.items[1].percentage).toBe(10);
      });

      it("single valid item has isEmpty=false and percentage=100", () => {
        const result = validateBarData([{ answer: "A", count: 42 }]);
        expect(result.isEmpty).toBe(false);
        expect(result.max).toBe(42);
        expect(result.items[0].percentage).toBe(100);
      });
    });
    ```

    ```typescript
    import { describe, it, expect } from "vitest";
    import { render } from "@testing-library/react";
    import { renderTooltipContent } from "../components/extraction/stacked-answer-bars";

    describe("renderTooltipContent", () => {
      it("returns null when active=false", () => {
        const result = renderTooltipContent({ active: false, payload: [] });
        expect(result).toBeNull();
      });

      it("returns null when payload is empty", () => {
        const result = renderTooltipContent({ active: true, payload: [] });
        expect(result).toBeNull();
      });

      it("shows integer count and percentage — never raw decimal", () => {
        const mockPayload = [
          {
            name: "grounded",
            value: 60,
            payload: { grounded: 60, best_effort: 15, insufficient_evidence: 4, extraction_error: 1 },
          },
        ];
        const element = renderTooltipContent({ active: true, payload: mockPayload });
        expect(element).not.toBeNull();
        // Render and check text does not contain raw decimal like "0.75"
        const { container } = render(element as React.ReactElement);
        expect(container.textContent).not.toMatch(/0\.\d{2}/);
        expect(container.textContent).toMatch(/60/); // count visible
        expect(container.textContent).toMatch(/%/);  // percentage sign visible
      });

      it("handles best_effort status correctly", () => {
        const mockPayload = [
          {
            name: "best_effort",
            value: 15,
            payload: { grounded: 60, best_effort: 15, insufficient_evidence: 4, extraction_error: 1 },
          },
        ];
        const element = renderTooltipContent({ active: true, payload: mockPayload });
        const { container } = render(element as React.ReactElement);
        expect(container.textContent).toMatch(/15/);
      });
    });
    ```
