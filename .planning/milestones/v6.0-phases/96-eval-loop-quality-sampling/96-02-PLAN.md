```typescript
"use node";

import { internalAction } from "../_generated/server";
import { internal } from "../_generated/api";
import { v } from "convex/values";
import { createOpenRouter } from "@openrouter/ai-sdk-provider";
import { generateText, Output } from "ai";
import { z } from "zod";
import { buildCriticPrompt } from "../../lib/ai/core/prompts";

const criticOutputSchema = z.object({
  verdict: z.enum(["supported", "unsupported", "partial"]).describe(
    "Whether the evidence quote appears verbatim in the source and supports the answer"
  ),
  reason: z.string().describe("Brief explanation of the verdict"),
});

export const sampleAndCheck = internalAction({
  args: {},
  returns: v.null(),
  handler: async (ctx) => {
    // 1. Get a random eligible result (completed, has criticVerdict, not sampled in last 7 days)
    const sample = await ctx.runQuery(
      internal.extraction.qualitySamplingHelpers.getRandomEligibleResult,
      {}
    );
    if (!sample) {
      // D-14: No eligible results -- early-return silently
      return null;
    }

    // 2. Load paper body text for critic re-run
    const paper = await ctx.runQuery(
      internal.extraction.qualitySamplingHelpers.getPaperBodyTextInternal,
      { paperId: sample.paperId }
    );
    if (!paper || !paper.bodyText) {
      console.info(JSON.stringify({
        event: "quality_sample_skip",
        reason: "no_body_text",
        answerId: String(sample._id),
      }));
      return null;
    }

    // 3. Load the question text for critic prompt
    const question = await ctx.runQuery(
      internal.extraction.qualitySamplingHelpers.getQuestionText,
      { jobId: sample.jobId, questionId: sample.questionId }
    );
    if (!question) {
      console.info(JSON.stringify({
        event: "quality_sample_skip",
        reason: "no_question",
        answerId: String(sample._id),
      }));
      return null;
    }

    // 4. Build critic prompt using existing infrastructure
    const sourceSlice = paper.bodyText.slice(
      sample.evidenceStartChar ?? 0,
      sample.evidenceEndChar ?? paper.bodyText.length
    );
    const criticPrompts = buildCriticPrompt({
      question: question,
      answer: sample.answer ?? "",
      evidenceQuote: sample.evidenceQuote ?? "",
      sourceSlice,
    });

    // 5. Run critic via OpenRouter (same model tier as runtime critic)
    const openrouter = createOpenRouter();
    const model = openrouter("google/gemini-2.5-flash-lite", {
      plugins: [{ id: "response-healing" }],
      provider: { allowFallbacks: true },
    });

    const judgeModel = "google/gemini-2.5-flash-lite";

    try {
      const { output: criticResult } = await generateText({
        model,
        output: Output.object({ schema: criticOutputSchema }),
        temperature: 0,
        system: criticPrompts.system,
        prompt: criticPrompts.prompt,
        abortSignal: AbortSignal.timeout(30_000),
      });

      const verdict = criticResult?.verdict ?? "partial";

      // 6. Determine reviewStatus per D-19
      const reviewStatus = verdict === "unsupported" ? "pending" : "reviewed";

      // 7. Write quality check via mutation
      await ctx.runMutation(
        internal.extraction.qualitySamplingHelpers.writeQualityCheck,
        {
          answerId: sample._id,
          verdict,
          judgeModel,
          timestamp: Date.now(),
          reviewStatus,
        }
      );

      console.info(JSON.stringify({
        event: "quality_sample_complete",
        answerId: String(sample._id),
        verdict,
        judgeModel,
        reviewStatus,
      }));
    } catch (error) {
      console.error(JSON.stringify({
        event: "quality_sample_error",
        answerId: String(sample._id),
        errorMessage: error instanceof Error ? error.message : String(error),
      }));
    }

    return null;
  },
});
```

```typescript
import { internalQuery, internalMutation } from "../_generated/server";
import { v } from "convex/values";

/**
 * Get a random eligible extraction result for quality sampling.
 * Eligible: has criticVerdict (completed critic pass), not already sampled in last 7 days.
 */
export const getRandomEligibleResult = internalQuery({
  args: {},
  returns: v.union(v.null(), v.object({
    _id: v.id("extractionResults"),
    jobId: v.id("extractionJobs"),
    paperId: v.id("extractionPapers"),
    questionId: v.string(),
    answer: v.union(v.string(), v.null()),
    evidenceQuote: v.optional(v.string()),
    evidenceStartChar: v.optional(v.number()),
    evidenceEndChar: v.optional(v.number()),
  })),
  handler: async (ctx) => {
    // Get all completed results with critic verdicts (these are eligible for sampling)
    const results = await ctx.db
      .query("extractionResults")
      .withIndex("by_job")
      .take(500);

    // Filter to results that have a criticVerdict (completed full pipeline)
    const eligible = results.filter((r) => r.criticVerdict != null && r.answer != null);
    if (eligible.length === 0) return null;

    // Check which have already been sampled in last 7 days
    const sevenDaysAgo = Date.now() - 7 * 24 * 60 * 60 * 1000;
    const recentChecks = await ctx.db
      .query("qualityChecks")
      .withIndex("by_timestamp", (q) => q.gte("timestamp", sevenDaysAgo))
      .take(1000);

    const recentlyCheckedIds = new Set(recentChecks.map((c) => String(c.answerId)));
    const unchecked = eligible.filter((r) => !recentlyCheckedIds.has(String(r._id)));

    if (unchecked.length === 0) return null;

    // Pick a random one
    const picked = unchecked[Math.floor(Math.random() * unchecked.length)];
    return {
      _id: picked._id,
      jobId: picked.jobId,
      paperId: picked.paperId,
      questionId: picked.questionId,
      answer: picked.answer,
      evidenceQuote: picked.evidenceQuote,
      evidenceStartChar: picked.evidenceStartChar,
      evidenceEndChar: picked.evidenceEndChar,
    };
  },
});

/**
 * Load paper body text for quality sampling critic re-run.
 */
export const getPaperBodyTextInternal = internalQuery({
  args: { paperId: v.id("extractionPapers") },
  returns: v.union(v.null(), v.object({ bodyText: v.union(v.string(), v.null()) })),
  handler: async (ctx, args) => {
    const paper = await ctx.db.get(args.paperId);
    if (!paper) return null;
    return { bodyText: paper.bodyText ?? null };
  },
});

/**
 * Load question text from the job's embedded questions array.
 */
export const getQuestionText = internalQuery({
  args: { jobId: v.id("extractionJobs"), questionId: v.string() },
  returns: v.union(v.null(), v.string()),
  handler: async (ctx, args) => {
    const job = await ctx.db.get(args.jobId);
    if (!job) return null;
    const q = job.questions.find((q) => q.key === args.questionId);
    return q?.question ?? null;
  },
});

/**
 * Write a quality check result (D-15, D-19).
 */
export const writeQualityCheck = internalMutation({
  args: {
    answerId: v.id("extractionResults"),
    verdict: v.string(),
    judgeModel: v.string(),
    timestamp: v.number(),
    reviewStatus: v.string(),
  },
  returns: v.null(),
  handler: async (ctx, args) => {
    await ctx.db.insert("qualityChecks", {
      answerId: args.answerId,
      verdict: args.verdict,
      judgeModel: args.judgeModel,
      timestamp: args.timestamp,
      reviewStatus: args.reviewStatus,
    });
    return null;
  },
});
```

```typescript
/**
 * 7-day rolling quality trend from production sampling (EVAL-05).
 *
 * Returns aggregate counts of supported/unsupported/partial verdicts
 * and a passRate (supported / total). Returns null for unauthenticated requests.
 */
export const getQualityTrend = query({
  args: {},
  returns: v.union(
    v.null(),
    v.object({
      supported: v.number(),
      unsupported: v.number(),
      partial: v.number(),
      total: v.number(),
      passRate: v.number(),
    })
  ),
  handler: async (ctx) => {
    const user = await getCurrentUser(ctx);
    if (!user) return null;

    const sevenDaysAgo = Date.now() - 7 * 24 * 60 * 60 * 1000;
    const checks = await ctx.db
      .query("qualityChecks")
      .withIndex("by_timestamp", (q) => q.gte("timestamp", sevenDaysAgo))
      .take(1000);

    let supported = 0;
    let unsupported = 0;
    let partial = 0;

    for (const check of checks) {
      if (check.verdict === "supported") supported++;
      else if (check.verdict === "unsupported") unsupported++;
      else partial++;
    }

    const total = checks.length;
    const passRate = total > 0 ? supported / total : 0;

    return { supported, unsupported, partial, total, passRate };
  },
});
```

**getReviewQueue** per D-20:
```typescript
/**
 * Pending human-review queue for quality-flagged extraction answers (EVAL-06).
 *
 * Returns quality checks with reviewStatus=pending, joined with answer
 * and paper data for review context. Bounded to 100 items.
 * Returns null for unauthenticated requests.
 */
export const getReviewQueue = query({
  args: {},
  returns: v.union(
    v.null(),
    v.array(
      v.object({
        qualityCheckId: v.id("qualityChecks"),
        answerId: v.id("extractionResults"),
        verdict: v.string(),
        judgeModel: v.string(),
        timestamp: v.number(),
        answer: v.union(v.string(), v.null()),
        question: v.union(v.string(), v.null()),
        evidenceQuote: v.union(v.string(), v.null()),
        paperTitle: v.union(v.string(), v.null()),
      })
    )
  ),
  handler: async (ctx) => {
    const user = await getCurrentUser(ctx);
    if (!user) return null;

    // Get pending review items, most recent first
    const pendingChecks = await ctx.db
      .query("qualityChecks")
      .withIndex("by_timestamp")
      .order("desc")
      .take(100);

    const pending = pendingChecks.filter((c) => c.reviewStatus === "pending");

    const items = [];
    for (const check of pending) {
      // Join with extraction result
      const result = await ctx.db.get(check.answerId);
      // Join with paper for title
      const paper = result ? await ctx.db.get(result.paperId) : null;
      // Get question text from job
      let questionText: string | null = null;
      if (result) {
        const job = await ctx.db.get(result.jobId);
        if (job) {
          const q = job.questions.find((q) => q.key === result.questionId);
          questionText = q?.question ?? null;
        }
      }

      items.push({
        qualityCheckId: check._id,
        answerId: check.answerId,
        verdict: check.verdict,
        judgeModel: check.judgeModel,
        timestamp: check.timestamp,
        answer: result?.answer ?? null,
        question: questionText,
        evidenceQuote: result?.evidenceQuote ?? null,
        paperTitle: paper?.title ?? null,
      });
    }

    return items;
  },
```
