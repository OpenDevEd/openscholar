**Submit handler** (per D-05):
```typescript
const createJob = useMutation(api.extraction.mutations.createExtractionJob);

const handleSubmit = useCallback(async () => {
  if (!papers || papers.length === 0 || localQuestions.length === 0) return;
  setIsSubmitting(true);
  try {
    const keyedQuestions = localQuestions.map((q, i) => ({
      key: `q_${i}`,
      question: q,
    }));
    await createJob({ papers, questions: keyedQuestions });
    showStyledToast("success", t("jobCreation.success"));
    onOpenChange(false);
    posthog.capture("extraction_job_created", {
      paperCount: papers.length,
      questionCount: localQuestions.length,
      tier,
    });
  } catch (error: unknown) {
    const msg = getErrorMessage(error, t("jobCreation.error"));
    if (
      typeof error === "object" && error !== null &&
      "data" in error &&
      typeof (error as { data?: unknown }).data === "string" &&
      ((error as { data: string }).data).includes("EXTRACTION_JOB_ALREADY_ACTIVE")
    ) {
      showStyledToast("error", t("jobCreation.activeJobWarning"));
    } else {
      showStyledToast("error", msg);
    }
  } finally {
    setIsSubmitting(false);
  }
}, [papers, localQuestions, tier, createJob, onOpenChange, t]);
```
