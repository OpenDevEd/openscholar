**Implementation:**
```typescript
"use client";

import React from "react";
import { FilterSidebar } from "@/components/shared/filter-sidebar";
import { useLibrarySidebarStore, getLibrarySidebarDimensions } from "@/lib/store";
import { useIsMobile } from "@/lib/hooks";
import { LibraryFilterContent } from "./library-filter-content";

export function LibraryFilterSidebar(props: LibraryFilterSidebarProps) {
  const isMobile = useIsMobile();

  const filterContent = (
    <LibraryFilterContent
      rawSearch={props.rawSearch}
      onSearchChange={props.onSearchChange}
      minRating={props.minRating}
      onMinRatingChange={props.onMinRatingChange}
      savedPreset={props.savedPreset}
      onSavedPresetChange={props.onSavedPresetChange}
      yearFrom={props.yearFrom}
      yearTo={props.yearTo}
      onYearFromChange={props.onYearFromChange}
      onYearToChange={props.onYearToChange}
      collectionId={props.collectionId}
      onCollectionChange={props.onCollectionChange}
      collections={props.collections}
      onManageCollections={props.onManageCollections}
      isMobile={isMobile}
    />
  );

  // Mobile: render raw filter content (SidebarLayout handles the tab chrome)
  // FilterSidebar returns null on mobile, so we must bypass it
  if (isMobile) {
    return (
      <div className="w-full max-w-2xl mx-auto">
        {filterContent}
      </div>
    );
  }

  // Desktop: wrap in FilterSidebar for glassmorphic chrome + collapse animation
  return (
    <FilterSidebar
      useStore={useLibrarySidebarStore}
      getDimensions={getLibrarySidebarDimensions}
      activeFilterCount={props.activeFilterCount}
      onClearAll={props.onClearAll}
    >
      {filterContent}
    </FilterSidebar>
  );
}
```

**Key details:**
- `useIsMobile` is imported from `@/lib/hooks` (same as SearchFilterSidebar)
- The mobile path renders raw `LibraryFilterContent` wrapped in `div.w-full.max-w-2xl.mx-auto` (matching SearchFilterSidebar)
- The desktop path wraps in `FilterSidebar` with `useLibrarySidebarStore` and `getLibrarySidebarDimensions` (from lib/store.ts)
- `activeFilterCount` and `onClearAll` are passed through to `FilterSidebar` -- the count drives the badge and the clear button
- Do NOT compute filter state inside this component -- it receives everything as props from the parent (`LibraryPageClient`)
