  <behavior>
    validateBarData:
    - null/undefined input → { isEmpty: true, max: 0, items: [] }
    - [] → { isEmpty: true, max: 0, items: [] }
    - All counts = 0 → { isEmpty: true, max: 0, items: [] }
    - NaN/negative counts filtered out; if nothing left → isEmpty: true
    - [{answer: "A", count: 90}, {answer: "B", count: 10}] →
      { isEmpty: false, max: 90, items: [
        { answer: "A", count: 90, percentage: 90 },
        { answer: "B", count: 10, percentage: 10 }
      ]}
    - Single valid item → { isEmpty: false, max: item.count, items: [{ percentage: 100 }] }

    SafeBarData type exported:
    { isEmpty: boolean; max: number; items: Array<{ answer: string; count: number; percentage: number }> }
  </behavior>
