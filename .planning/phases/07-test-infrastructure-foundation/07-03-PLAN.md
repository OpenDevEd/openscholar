import type { Page, Locator } from "@playwright/test";

/**
 * Page Object Model for the billing settings page (/settings/billing).
 * All selectors use data-testid attributes added in Plan 01.
 */
export class BillingPage {
  readonly page: Page;

  // Locators (lazy, evaluated on access)
  readonly planName: Locator;
  readonly statusBadge: Locator;
  readonly renewalDate: Locator;
  readonly pendingChangeBanner: Locator;
  readonly keepPlanButton: Locator;
  readonly creditCapBanner: Locator;
  readonly manageButton: Locator;
  readonly creditBalance: Locator;
  readonly freeCredits: Locator;
  readonly paidCredits: Locator;

  constructor(page: Page) {
    this.page = page;
    this.planName = page.locator('[data-testid="billing-plan-name"]');
    this.statusBadge = page.locator('[data-testid="billing-status-badge"]');
    this.renewalDate = page.locator('[data-testid="billing-renewal-date"]');
    this.pendingChangeBanner = page.locator('[data-testid="billing-pending-change-banner"]');
    this.keepPlanButton = page.locator('[data-testid="billing-keep-plan-button"]');
    this.creditCapBanner = page.locator('[data-testid="billing-credit-cap-banner"]');
    this.manageButton = page.locator('[data-testid="billing-manage-button"]');
    this.creditBalance = page.locator('[data-testid="billing-credit-balance"]');
    this.freeCredits = page.locator('[data-testid="billing-free-credits"]');
    this.paidCredits = page.locator('[data-testid="billing-paid-credits"]');
  }

  /**
   * Navigate to the billing page and wait for the plan name to be visible.
   * This ensures the page has loaded and subscription state has settled.
   */
  async goto(): Promise<void> {
    await this.page.goto("/settings/billing");
    await this.planName.waitFor({ state: "visible", timeout: 30_000 });
  }

  /**
   * Get the displayed plan name (e.g., "Go Plan", "Pro Plan", "Free Plan").
   */
  async getPlanName(): Promise<string> {
    return (await this.planName.textContent()) ?? "";
  }

  /**
   * Get the subscription status badge text (e.g., "Active", "Inactive").
   */
  async getStatusBadgeText(): Promise<string> {
    return (await this.statusBadge.textContent()) ?? "";
  }

  /**
   * Get the renewal/expiry date text (e.g., "Renews on Mar 15, 2026").
   * Returns empty string if no renewal date is shown (free plan).
   */
  async getRenewalText(): Promise<string> {
    const count = await this.renewalDate.count();
    if (count === 0) return "";
    return (await this.renewalDate.textContent()) ?? "";
  }

  /**
   * Get the total credit balance as a number.
   * Strips non-digit characters (commas, spaces) and parses as integer.
   * Returns NaN if the element is not visible or text is not parseable.
   */
  async getCreditBalance(): Promise<number> {
    const count = await this.creditBalance.count();
    if (count === 0) return NaN;
    const text = (await this.creditBalance.textContent()) ?? "";
    return parseInt(text.replace(/[^0-9-]/g, ""), 10);
  }

  /**
   * Click the manage subscription / upgrade button.
   */
  async clickManageButton(): Promise<void> {
    await this.manageButton.click();
  }

  /**
   * Click the "Keep current plan" button to cancel a scheduled downgrade.
   */
  async clickKeepPlan(): Promise<void> {
    await this.keepPlanButton.click();
  }
}

import type { Page, Locator } from "@playwright/test";

type PlanId = "free" | "go" | "pro" | "enterprise";

/**
 * Page Object Model for the pricing page (/pricing).
 * All selectors use data-testid attributes added in Plan 01.
 */
export class PricingPage {
  readonly page: Page;

  constructor(page: Page) {
    this.page = page;
  }

  /**
   * Navigate to the pricing page and wait for plan cards to be visible.
   */
  async goto(): Promise<void> {
    await this.page.goto("/pricing");
    // Wait for at least one plan card to be visible
    await this.page.locator('[data-testid^="pricing-plan-"]').first().waitFor({
      state: "visible",
      timeout: 30_000,
    });
  }

  /**
   * Get the locator for a specific plan card.
   */
  planCard(planId: PlanId): Locator {
    return this.page.locator(`[data-testid="pricing-plan-${planId}"]`);
  }

  /**
   * Get the locator for a specific plan's action button.
   */
  planButton(planId: PlanId): Locator {
    return this.page.locator(`[data-testid="pricing-button-${planId}"]`);
  }

  /**
   * Click the subscribe/action button for a specific plan.
   */
  async clickSubscribe(planId: PlanId): Promise<void> {
    await this.planButton(planId).click();
  }

  /**
   * Get the text of a plan's action button (e.g., "Upgrade to Pro", "Current Plan").
   */
  async getButtonText(planId: PlanId): Promise<string> {
    return (await this.planButton(planId).textContent()) ?? "";
  }

  /**
   * Get the "Current Plan" badge locator.
   * Use with expect(pricingPage.currentBadge).toBeVisible() to check if a plan is current.
   */
  get currentBadge(): Locator {
    return this.page.locator('[data-testid="pricing-current-badge"]');

   * Get the "Scheduled" badge locator.
   * Use with expect(pricingPage.scheduledBadge).toBeVisible() to check if a plan change is scheduled.

  get scheduledBadge(): Locator {

   * Get the upgrade confirmation dialog's confirm button locator.

  /**

**Implementation:**

```typescript
import { test as base, expect } from "@playwright/test";
import { createClerkClient } from "@clerk/backend";
import { clerk } from "@clerk/testing/playwright";
import Stripe from "stripe";
import { createStripeClient, cleanupStripeCustomer } from "./stripe-helpers";
import { WebhookWaiter } from "./webhook-waiter";
import { BillingPage } from "./pages/billing-page";
import { PricingPage } from "./pages/pricing-page";

/**
 * Shape of the Clerk test user created by the fixture.
 */
type ClerkTestUser = {
  id: string;
  email: string;
  password: string;
};

/**
 * Fixtures provided to every billing E2E test.
 */
type BillingFixtures = {
  /** Fresh Clerk user, signed in, auto-cleaned on teardown */
  clerkUser: ClerkTestUser;
  /** Pre-configured Stripe SDK client */
  stripeClient: Stripe;
  /** Webhook waiter bound to the test user's Convex ID */
  webhookWaiter: WebhookWaiter;
  /** BillingPage POM instance */
  billingPage: BillingPage;
  /** PricingPage POM instance */
  pricingPage: PricingPage;
};

/**
 * Billing test fixture that composes user lifecycle, Stripe client,
 * webhook waiter, and page objects into a single test extension.
 *
 * Usage in Phase 8/9 specs:
 * ```typescript
 * import { test, expect } from "../fixtures";
 *
 * test("user subscribes to Go plan", async ({ clerkUser, stripeClient, webhookWaiter, billingPage }) => {
 *   // ... test body (~10-15 lines)
 * });
 * ```
 */
export const test = base.extend<BillingFixtures>({
  // --- clerkUser fixture ---
  // Creates a fresh Clerk user per test, signs in, and cleans up on teardown.
  clerkUser: async ({ page }, use) => {
    const secretKey = process.env.CLERK_SECRET_KEY;
    if (!secretKey) {
      throw new Error("Missing CLERK_SECRET_KEY for billing E2E tests");
    }

    const clerkClient = createClerkClient({ secretKey });

    // Generate unique email per test to ensure isolation
    const uniqueId = `${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
    const email = `billing-e2e-${uniqueId}@test.example.com`;
    const password = `E2EBilling${uniqueId}!`;

    // Create the Clerk user
    const user = await clerkClient.users.createUser({
      emailAddress: [email],
      password,
      skipPasswordChecks: true,
      skipPasswordRequirement: true,
      firstName: "E2E",
      lastName: "Billing",
    });

    // Sign in to the browser session
    await page.goto("/");
    await clerk.loaded({ page });
    await clerk.signIn({
      page,
      signInParams: {
        strategy: "password",
        identifier: email,
        password,
      },
    });

    // Navigate to a protected page to trigger ensureUser in Convex
    // This creates the Convex user row via the Clerk webhook
    await page.goto("/settings/billing");
    // Wait for the page to finish loading (subscription state settles)
    await page.waitForLoadState("networkidle");

    await use({ id: user.id, email, password });

    // Teardown: sign out and delete the Clerk user
    try {
      await clerk.signOut({ page });
    } catch {
      // Best-effort -- page may have navigated away
    }
    await clerkClient.users.deleteUser(user.id);
  },

  // --- stripeClient fixture ---
  stripeClient: async ({}, use) => {
    const stripe = createStripeClient();
    await use(stripe);
  },

  // --- webhookWaiter fixture ---
  // Waits for clerkUser to be available, discovers Convex userId,
  // creates WebhookWaiter bound to that user, and resets state on teardown.
  webhookWaiter: async ({ clerkUser }, use) => {
    // Discover the Convex userId by querying with the Clerk user identity.
    // The Convex user is created by the ensureUser mutation triggered during
    // page navigation in the clerkUser fixture.
    //
    // We use convex run with an identity to call ensureUser which returns
    // the Convex userId.
    const { execFile } = await import("node:child_process");
    const { promisify } = await import("node:util");
    const execFileAsync = promisify(execFile);

    const identity = {
      subject: clerkUser.id,
      email: clerkUser.email,
      givenName: "E2E",
      familyName: "Billing",
    };

    const cliArgs = [
      "convex",
      "run",
      "users/mutations:ensureUser",
      JSON.stringify({}),
      "--identity",
      JSON.stringify(identity),
      "--typecheck",
      "disable",
      "--codegen",
      "disable",
    ];

    if (process.env.CONVEX_DEPLOYMENT_NAME) {
      cliArgs.push("--deployment-name", process.env.CONVEX_DEPLOYMENT_NAME);
    }

    const { stdout } = await execFileAsync("npx", cliArgs, {
      env: process.env,
      maxBuffer: 1024 * 1024,
      timeout: 30_000,
    });

    const convexUserId = stdout.trim().split("\n").at(-1)?.trim();
    if (!convexUserId) {
      throw new Error("Failed to discover Convex userId for billing test user");
    }

    // Parse the userId -- may be JSON-quoted string
    let parsedUserId: string;
    try {
      parsedUserId = JSON.parse(convexUserId);
    } catch {
      parsedUserId = convexUserId;
    }

    const waiter = new WebhookWaiter(parsedUserId);
    await use(waiter);

    // Teardown: reset billing state in Convex
    try {
      await waiter.resetBillingState();
    } catch {
      // Best-effort cleanup
    }
  },

  // --- billingPage fixture ---
  billingPage: async ({ page }, use) => {
    await use(new BillingPage(page));
  },

  // --- pricingPage fixture ---
  pricingPage: async ({ page }, use) => {
    await use(new PricingPage(page));
  },
});

export { expect };
```

Key design decisions:
- **Per-test user isolation**: Each test gets a unique Clerk user with timestamped email. No cross-test contamination (per research pitfall #2).
- **`signInParams` with strategy/password**: Uses the correct `@clerk/testing` v1.x API (per research State of the Art table).
- **Convex user discovery**: After Clerk sign-in and page navigation, uses `convex run users/mutations:ensureUser` with identity to get the Convex userId. This resolves Research Open Question #1.
- **Layered teardown**: clerkUser teardown deletes Clerk user. webhookWaiter teardown resets Convex billing state. Stripe cleanup happens in individual tests after `createSubscriptionBypass` (since not all tests create subscriptions).
- **Page navigation in clerkUser fixture**: Navigates to `/settings/billing` to trigger Convex `ensureUser` and wait for subscription state to settle. This addresses Research Pitfall #4 (Convex user not synced from Clerk webhook).
