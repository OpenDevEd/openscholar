**4. DOI resolution handlers (D-09, D-10):**
```typescript
const handleResolve = useCallback(async (itemKey: string, doi: string) => {
  const cache = resolvedCacheRef.current;

  // Check cache first (D-10)
  if (cache.has(doi)) {
    const cachedId = cache.get(doi);
    if (cachedId) {
      // Find the item for its title
      const item = items.find(i => i.key === itemKey);
      if (item) {
        addToBasket({ rawId: cachedId, title: item.title, doi: doi, source: "zotero" });
        setItemStates(prev => new Map(prev).set(itemKey, "resolved"));
      }
    } else {
      // Cached as null (previously failed) -- show failed state
      setItemStates(prev => new Map(prev).set(itemKey, "failed"));
    }
    return;
  }

  setItemStates(prev => new Map(prev).set(itemKey, "resolving"));
  try {
    const result = await resolveByDoi({ doi });
    if (result) {
      cache.set(doi, result.openalexId);
      addToBasket({ rawId: result.openalexId, title: result.title, doi: result.doi ?? null, source: "zotero" });
      setItemStates(prev => new Map(prev).set(itemKey, "resolved"));
    } else {
      cache.set(doi, null);
      setItemStates(prev => new Map(prev).set(itemKey, "failed"));
    }
  } catch {
    cache.set(doi, null);
    setItemStates(prev => new Map(prev).set(itemKey, "failed"));
  }
}, [resolveByDoi, addToBasket, items]);

const handleRetry = useCallback((itemKey: string, doi: string) => {
  // Clear cache entry for this DOI before retrying
  resolvedCacheRef.current.delete(doi);
  setItemStates(prev => new Map(prev).set(itemKey, "idle"));
  handleResolve(itemKey, doi);
}, [handleResolve]);
```
