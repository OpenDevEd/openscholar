```typescript
export function ZoteroItemRow({ item, resolutionState, onResolve, onRetry, isInBasket }: ZoteroItemRowProps) {
  const t = useTranslations("extraction.zoteroTab");

  // State machine for the checkbox/indicator column
  // Priority: no DOI > in basket (resolved) > resolving > failed > idle
  function renderCheckboxArea() {
    if (!item.doi) {
      // STATE 5: No DOI — disabled checkbox with tooltip (ZOT-04, D-08)
      return (
        <Tooltip>
          <TooltipTrigger asChild>
            <span>
              <Checkbox
                disabled
                aria-disabled="true"
                aria-label={t("noDoi")}
              />
            </span>
          </TooltipTrigger>
          <TooltipContent side="right">{t("noDoi")}</TooltipContent>
        </Tooltip>
      );
    }

    if (isInBasket) {
      // STATE 3: Resolved + in basket — green checkmark
      return (
        <Check
          className="h-4 w-4 text-green-600 dark:text-green-400"
          aria-label={t("resolved")}
        />
      );
    }

    if (resolutionState === "resolving") {
      // STATE 2: Resolving — inline spinner replacing checkbox
      return (
        <Loader2
          className="h-4 w-4 animate-spin text-muted-foreground"
          aria-label={t("resolving")}
        />
      );
    }

    if (resolutionState === "failed") {
      // STATE 4: Failed — red X with retry button and tooltip
      return (
        <Tooltip>
          <TooltipTrigger asChild>
            <button
              onClick={() => onRetry(item.key, item.doi!)}
              className="rounded p-1 -m-1 focus:outline-none focus:ring-2 focus:ring-ring"
              aria-label={t("retryResolve")}
            >
              <X className="h-4 w-4 text-destructive cursor-pointer" />
            </button>
          </TooltipTrigger>
          <TooltipContent side="right">{t("resolveFailed")}</TooltipContent>
        </Tooltip>
      );
    }

    // STATE 1: Idle — unchecked checkbox ready to resolve
    return (
      <Checkbox
        checked={false}
        onCheckedChange={() => onResolve(item.key, item.doi!)}
        aria-label={`Add ${item.title} to extraction`}
      />
    );
  }

  return (
    <TooltipProvider>
      <div
        className={cn(
          "flex items-center gap-3 py-3",
          (isInBasket || !item.doi) && "opacity-60"
        )}
      >
        {/* Checkbox / indicator column — 24px fixed width (UI spec) */}
        <div className="flex-shrink-0 w-6 flex items-center justify-center">
          {renderCheckboxArea()}
        </div>

        {/* Content */}
        <div className="min-w-0 flex-1">
          <p className="text-sm font-medium truncate">{item.title}</p>
          {item.authors.length > 0 && (
            <p className="text-xs text-muted-foreground truncate">
              {item.authors.join(", ")}
            </p>
          )}
          <div className="flex flex-wrap items-center gap-1 mt-1">
            {item.year && (
              <span className="text-xs text-muted-foreground">{item.year}</span>
            )}
            <Badge variant="secondary" className="text-[10px] px-1.5 py-0">
              {item.itemType}
            </Badge>
            {item.doi && (
              <Badge variant="outline" className="text-[10px] px-1.5 py-0 font-mono">
                {item.doi.length > 30 ? `${item.doi.slice(0, 30)}…` : item.doi}
              </Badge>
            )}
            {item.hasPdfChild && (
              <Badge variant="secondary" className="text-[10px] px-1.5 py-0">
                <FileText className="w-2.5 h-2.5 mr-0.5" />
                {t("hasPdf")}
              </Badge>
            )}
            {isInBasket && (
              <span className="text-xs text-muted-foreground ml-auto">
                {t("alreadyInBasket")}
              </span>
            )}
          </div>
        </div>
      </div>
    </TooltipProvider>
  );
}
```

**DOI resolution handlers (Section 4) — D-09, D-10:**
```typescript
const handleResolve = useCallback(
  async (itemKey: string, doi: string) => {
    const cache = resolvedCacheRef.current;

    // Check cache first (D-10) — cache key is DOI, not item key (Pitfall 5)
    if (cache.has(doi)) {
      const cachedId = cache.get(doi);
      if (cachedId) {
        const item = items.find((i) => i.key === itemKey);
        if (item) {
          addToBasket({ rawId: cachedId, title: item.title, doi, source: "zotero" });
          setItemStates((prev) => new Map(prev).set(itemKey, "resolved"));
        }
      } else {
        // Previously failed — show failed state without another API call
        setItemStates((prev) => new Map(prev).set(itemKey, "failed"));
      }
      return;
    }

    // Not in cache — call resolveByDoi
    setItemStates((prev) => new Map(prev).set(itemKey, "resolving"));
    try {
      const result = await resolveByDoi({ doi });
      if (result) {
        cache.set(doi, result.openalexId);
        addToBasket({
          rawId: result.openalexId,
          title: result.title,
          doi: result.doi ?? null,
          source: "zotero",
        });
        setItemStates((prev) => new Map(prev).set(itemKey, "resolved"));
      } else {
        cache.set(doi, null); // null = not found in OpenAlex
        setItemStates((prev) => new Map(prev).set(itemKey, "failed"));
      }
    } catch {
      cache.set(doi, null);
      setItemStates((prev) => new Map(prev).set(itemKey, "failed"));
    }
  },
  [resolveByDoi, addToBasket, items],
);

const handleRetry = useCallback(
  (itemKey: string, doi: string) => {
    // Clear the cached failure before retrying (D-10)
    resolvedCacheRef.current.delete(doi);
    setItemStates((prev) => new Map(prev).set(itemKey, "idle"));
    void handleResolve(itemKey, doi);
  },
  [handleResolve],
```
