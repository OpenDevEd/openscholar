export interface ExportOptions {
  includeMetadata?: boolean;     // Default: true in exportToCSV, not used in exportToJSON
  includeConfidence?: boolean;   // Default: true in exportToCSV, not used in exportToJSON
  includeReasoning?: boolean;    // Default: true in exportToJSON, not used in exportToCSV
  includeCitations?: boolean;    // Default: true in exportToJSON, not used in exportToCSV
}

export function exportToCSV(
  papers: PaperWithResults[],
  questions: Question[],
  options?: ExportOptions
): void
// Builds CSV with:
// - Always: Paper Title, DOI, Year
// - If includeMetadata (default true): Authors, Open Access, PDF Used
// - For each question: answer column
// - If includeConfidence (default true): confidence % column per question

  questions: Question[],
  options?: ExportOptions
): void

// - Calls downloadFile(json, filename, 'application/json')
