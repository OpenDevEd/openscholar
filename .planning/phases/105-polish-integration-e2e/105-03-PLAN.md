    import { describe, it, expect } from "vitest";
    import enUS from "@/messages/en-US.json";
    import enGB from "@/messages/en-GB.json";
    import fr from "@/messages/fr.json";
    import hi from "@/messages/hi.json";
    import ja from "@/messages/ja.json";
    import zh from "@/messages/zh.json";

    /** Flattens a JSON object into sorted dot-path keys.
     * Treats arrays + primitives + null as leaves (the path to them is the key). */
    function flatten(obj: unknown, prefix = ""): string[] {
      if (obj === null || typeof obj !== "object") return [prefix];
      if (Array.isArray(obj)) return [prefix];
      const keys: string[] = [];
      for (const [k, v] of Object.entries(obj as Record<string, unknown>)) {
        const path = prefix ? `${prefix}.${k}` : k;
        if (v !== null && typeof v === "object" && !Array.isArray(v)) {
          keys.push(...flatten(v, path));
        } else {
          keys.push(path);
        }
      }
      return keys.sort();
    }

    const LOCALES: Record<string, unknown> = {
      "en-US": enUS,
      "en-GB": enGB,
      fr,
      hi,
      ja,
      zh,
    };

    /** Walks a dot-path inside a locale object; returns undefined if missing. */
    function walk(obj: unknown, path: string): unknown {
      return path.split(".").reduce<unknown>(
        (acc, k) => (acc as Record<string, unknown> | null | undefined)?.[k],
        obj,
      );
    }

     * Each value in fr.json for these paths MUST contain at least one French diacritic.
     * D-25 mandates 8-10 samples — bounds encoded below as tripwires. */

      const cliArgs = [

        "disable",
        "--codegen",

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| E2E runner → Convex deployment | The seed action `npx convex run` invocation sends a command from the test runner to a Convex backend. In dev this is localhost; in CI/preview it's a dedicated preview deployment. Trust boundary is the network call. |
| Convex deployment env → `assertDevEnvironment()` | The guard inspects `CONVEX_CLOUD_URL`, `NODE_ENV`, `ALLOW_TEST_SEEDING`. These env vars are set server-side — untamperable from the client. The risk surface is a compromised deployment key. |
| Seed mutation → `E2E_CLERK_USER_EMAIL` scoped userId | The mutation looks up a user strictly by email env var; throws on miss. No fallback widens the blast radius. |
| Playwright `page.route()` → real network | Intercepted routes NEVER hit zotero.org. Non-intercepted routes proceed normally (Clerk, Convex). The fixture JSONs are committed under VCS. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-105-10 | Elevation of Privilege | `seedCompletedJob` callable in production | mitigate | Triple-condition `assertDevEnvironment()` check (CONVEX_CLOUD_URL localhost OR NODE_ENV=development OR ALLOW_TEST_SEEDING=true) — same pattern as `convex/lib/env.ts:requireAppUrl`. Action is `internalAction` not `action`, so it cannot be reached from the public API even if a client tried to call it directly. Defense in depth. |
| T-105-11 | Tampering | Seed action inserts arbitrary data into production tables if guard is bypassed | mitigate | The seed writes only to `extractionJobs`/`extractionPapers`/`extractionResults`/`extractionQuestions` under the user matched by `E2E_CLERK_USER_EMAIL`. Mutation throws on empty or non-matching email — no silent attachment to a random user. Even if the triple dev-guard is bypassed, collateral is scoped to one user's job history. Fully reversible by deleting the test user's completed jobs. |
| T-105-12 | Information Disclosure | Zotero API fixture JSON committed to repo exposes library IDs | accept | `library.id: 123456` is a synthetic test value, not a real user's library ID. `key` fields (ABCD1234, ITEM1111, etc.) are fabricated per the fixture. No PII. |
| T-105-13 | Denial of Service | Seed inserts `paperCount * questionCount` rows per run; bulk calls could overwhelm the deployment | accept | Typical test call is `paperCount: 10, questionCount: 3` → 30 extractionResults rows + 10 extractionPapers + 1 extractionJob = 41 rows. Convex handles this in <50ms. Even pathological calls (paperCount: 1000) stay within Convex's transaction limits because the mutation commits all inserts atomically. Attacker can't induce DoS without first compromising dev access. |
| T-105-14 | Spoofing | Tests use Clerk test user — if creds leak, an attacker could invoke seed via CLI | mitigate | `E2E_CLERK_USER_EMAIL` is loaded from `.env.local` (not committed) and from CI secrets. The test user has Clerk test-mode auth (not production) and cannot access real user data. Production Clerk deployment rejects test-mode tokens. |
| T-105-15 | Tampering | `page.route()` mock returns attacker-controlled JSON if fixtures are modified | mitigate | Fixture JSONs (`e2e/fixtures/zotero-*.json`) are committed under VCS. PR reviewers see diffs. Modifications go through normal code review. Note: the mocks ONLY apply during Playwright test execution — they cannot affect production runtime. |
| T-105-16 | Information Disclosure | i18n parity test output lists every missing/extra key on failure — could leak locale contents in CI logs | accept | i18n key paths (e.g., `extraction.workbench.someFeature`) are not secrets. The VALUES are not printed, only the PATHS. Standard practice in every i18n CI pipeline. |

**Security posture:** Low-medium threat surface. The seed action is the primary risk, but it's: (a) `internalAction` (never public), (b) guarded by a triple-condition dev check, (c) writes only to the user matched by `E2E_CLERK_USER_EMAIL` (throws on miss — no silent fallback). The Zotero mocks and E2E flow add zero new attack vectors. The i18n parity test is pure file-read + set comparison (no attack surface).
