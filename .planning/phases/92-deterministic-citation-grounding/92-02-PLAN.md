    **2. Create `components/extraction/evidence-quote.tsx`.**

    Per UI-SPEC section "1. EvidenceQuote":

    ```typescript
    "use client";

    import { useQuery } from "convex/react";
    import { useTranslations } from "next-intl";
    import { AlertCircle } from "lucide-react";
    import type { Id } from "@/convex/_generated/dataModel";
    import { api } from "@/convex/_generated/api";
    import { cn } from "@/lib/utils";

    interface EvidenceQuoteProps {
      paperId: Id<"extractionPapers">;
      startChar: number;
      endChar: number;
      className?: string;
    }

    export function EvidenceQuote({ paperId, startChar, endChar, className }: EvidenceQuoteProps) {
      const t = useTranslations("extraction");
      const paperData = useQuery(api.extraction.queries.getPaperBodyText, { paperId });

      // Empty range -- parent should not render this component
      if (startChar === endChar || startChar === undefined || endChar === undefined) {
        return null;
      }

      // Loading state (undefined = query pending)
      if (paperData === undefined) {
        return (
          <div className={cn("space-y-2 rounded-md p-2", className)}>
            <div className="h-[14px] w-full animate-pulse rounded bg-muted" />
            <div className="h-[14px] w-[85%] animate-pulse rounded bg-muted" />
            <div className="h-[14px] w-[60%] animate-pulse rounded bg-muted" />
          </div>
        );
      }

      // Error: paper not found or no access
      if (paperData === null || !paperData.bodyText) {
        return (
          <div className={cn("flex items-center gap-2 text-sm text-muted-foreground", className)}>
            <AlertCircle className="h-4 w-4 shrink-0" aria-hidden="true" />
            <span>{t("evidenceUnavailable")}</span>
          </div>
        );
      }

      // Out-of-bounds check
      if (startChar < 0 || endChar > paperData.bodyText.length || startChar >= endChar) {
        return (
          <div className={cn("flex items-center gap-2 text-sm text-muted-foreground", className)}>
            <AlertCircle className="h-4 w-4 shrink-0" aria-hidden="true" />
            <span>{t("evidenceUnavailable")}</span>
          </div>
        );
      }

      // Render verbatim slice -- NEVER use LLM evidenceQuote
      const verbatimText = paperData.bodyText.slice(startChar, endChar);

      return (
        <div
          role="link"
          tabIndex={0}
          className={cn(
            "cursor-pointer rounded-md border-l-[3px] border-l-primary/40 bg-[var(--pdf-highlight)] px-4 py-2 text-sm leading-[1.43] motion-safe:transition-colors",
            "hover:bg-[color-mix(in_oklch,var(--pdf-highlight)_70%,transparent)]",
            "focus-visible:outline-none focus-visible:ring-[3px] focus-visible:ring-ring/50",
            className
          )}
          aria-label={`View evidence in source at characters ${startChar} to ${endChar}`}
        >
          {verbatimText}
        </div>
      );
    }
    ```

    Key requirements per EXTR-04 and D-14:
    - Fetches bodyText via `useQuery(api.extraction.queries.getPaperBodyText, { paperId })`
    - Renders `paperData.bodyText.slice(startChar, endChar)` -- NEVER any LLM field
    - Three-state: loading skeleton, error, rendered quote
    - Left border 3px `var(--primary)` at 40% opacity
    - Background `var(--pdf-highlight)`
    - `role="link"`, `tabIndex={0}`, `aria-label` per UI-SPEC accessibility
    - `motion-safe:transition-colors` per reduced-motion accessibility
    - `focus-visible:ring-[3px] focus-visible:ring-ring/50` per existing shadcn pattern

    ```typescript
    "use client";

    import { useTranslations } from "next-intl";
    import { Badge } from "@/components/ui/badge";
    import {
      Tooltip,
      TooltipContent,
      TooltipTrigger,
    } from "@/components/ui/tooltip";
    import { cn } from "@/lib/utils";

    type CriticVerdict = "supported" | "unsupported" | "partial" | "timeout";

    interface CriticVerdictBadgeProps {
      verdict: CriticVerdict | null | undefined;
      reason?: string;
      className?: string;
    }

    const verdictConfig = {
      supported: {
        i18nKey: "criticVerified" as const,
        classes: "ai-surface",
      },
      unsupported: {
        i18nKey: "criticUnverified" as const,
        classes: "bg-[oklch(0.95_0.06_27/0.15)] border-[oklch(0.6_0.15_27/0.3)] text-[oklch(0.5_0.15_27)] dark:bg-[oklch(0.3_0.08_27/0.2)] dark:border-[oklch(0.6_0.15_27/0.3)] dark:text-[oklch(0.7_0.12_27)]",
      },
      partial: {
        i18nKey: "criticPartial" as const,
        classes: "bg-[oklch(0.95_0.06_85/0.15)] border-[oklch(0.6_0.1_85/0.3)] text-[oklch(0.5_0.08_85)] dark:bg-[oklch(0.3_0.08_85/0.2)] dark:border-[oklch(0.6_0.1_85/0.3)] dark:text-[oklch(0.7_0.08_85)]",
      },
      timeout: {
        i18nKey: "criticTimeout" as const,
        classes: "bg-muted border-border text-muted-foreground",
      },
    } as const;

    export function CriticVerdictBadge({ verdict, reason, className }: CriticVerdictBadgeProps) {
      const t = useTranslations("extraction");

      // Don't render if no verdict
      if (!verdict) return null;

      const config = verdictConfig[verdict];
      if (!config) return null;

      const badge = (
        <Badge
          variant="outline"
          className={cn(
            "rounded-full text-xs font-semibold",
            config.classes,
            className
          )}
        >
          {t(config.i18nKey)}
        </Badge>
      );

      // Wrap in tooltip if reason is available
      if (reason) {
        return (
          <Tooltip delayDuration={300}>
            <TooltipTrigger asChild>
              {badge}
            </TooltipTrigger>
            <TooltipContent>
              <p className="max-w-xs text-sm">{reason}</p>
            </TooltipContent>
          </Tooltip>
        );
      }

      return badge;
    }
    ```

    - Red semantic for `unsupported`, amber for `partial`, neutral muted for `timeout`
    - OKLCH colors with dark mode variants
    - Tooltip on hover/focus showing reason text, 300ms delay per UI-SPEC interaction contract
    - `text-xs font-semibold rounded-full` matching ConfidenceBadge
