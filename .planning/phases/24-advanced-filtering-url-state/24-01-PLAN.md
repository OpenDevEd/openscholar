   ```typescript
   export function savedPresetToTimestamp(preset: string | null): number | undefined {
     if (!preset) return undefined;
     const now = new Date();
     switch (preset) {
       case "week": return subWeeks(now, 1).getTime();
       case "month": return subMonths(now, 1).getTime();
       case "year": return subYears(now, 1).getTime();
       default: return undefined;
     }
   }
   ```

<task type="auto">
  <name>Task 2: Extend Convex queries with date and year filter arguments</name>
  <files>convex/library/queries.ts</files>
  <action>
Extend both `browseLibrary` and `searchLibrary` queries in `convex/library/queries.ts` to accept three new optional arguments:

1. **Add to args** (both queries):
   - `savedAfter: v.optional(v.number())` -- Unix timestamp threshold for date saved filter
   - `yearFrom: v.optional(v.number())` -- publication year lower bound (inclusive)
   - `yearTo: v.optional(v.number())` -- publication year upper bound (inclusive)

2. **Replace the existing single-predicate filter** with a composed multi-predicate filter. The current code only handles `minRating`:

   ```typescript
   // BEFORE (single predicate):
   if (args.minRating !== undefined) {
     const minRating = args.minRating;
     return await filter(q, (item) => item.rating >= minRating).paginate(args.paginationOpts);
   }
   return await q.paginate(args.paginationOpts);
   ```

   Replace with:

   ```typescript
   // AFTER (composed multi-predicate):
   const needsFilter =
     args.minRating !== undefined ||
     args.savedAfter !== undefined ||
     args.yearFrom !== undefined ||
     args.yearTo !== undefined;

   if (needsFilter) {
     return await filter(q, (item) => {
       if (args.minRating !== undefined && item.rating < args.minRating) return false;
       if (args.savedAfter !== undefined && item.savedAt < args.savedAfter) return false;
       if (args.yearFrom !== undefined && (item.publicationYear == null || item.publicationYear < args.yearFrom)) return false;
       if (args.yearTo !== undefined && (item.publicationYear == null || item.publicationYear > args.yearTo)) return false;
       return true;
     }).paginate(args.paginationOpts);
   }

   return await q.paginate(args.paginationOpts);
   ```

**Critical:** The year range filter MUST handle `publicationYear == null` explicitly -- items without a publication year are excluded when a year filter is active (they have no year to compare). Use `item.publicationYear == null` (double equals) to catch both null and undefined.

Apply this same pattern to BOTH `browseLibrary` AND `searchLibrary`. The `getLibraryCount` query does NOT need these filters (it always returns total count).

Do NOT add new indexes. The existing indexes (`by_user_savedAt`, `by_user_rating`, `by_user_year`) are sufficient. The `convex-helpers/server/filter` function handles the multi-predicate composition.
