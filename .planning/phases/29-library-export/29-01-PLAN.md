**1. Add `exportLibrary` query to `convex/library/queries.ts`:**

Add a new non-paginated query after `getLibraryCount`. This query replicates the exact filter logic from `browseLibrary`/`searchLibrary` but uses `.collect()` instead of `.paginate()`.

```typescript
export const exportLibrary = query({
  args: {
    sortBy: v.union(v.literal("savedAt"), v.literal("rating"), v.literal("publicationYear")),
    sortOrder: v.union(v.literal("asc"), v.literal("desc")),
    minRating: v.optional(v.number()),
    savedAfter: v.optional(v.number()),
    yearFrom: v.optional(v.number()),
    yearTo: v.optional(v.number()),
    collectionId: v.optional(v.id("collections")),
    searchText: v.optional(v.string()),
  },
  handler: async (ctx, args) => {
    const user = await requireAuth(ctx);

    // Pre-fetch collection member IDs if filtering by collection
    let collectionItemIds: Set<string> | null = null;
    if (args.collectionId) {
      const collection = await ctx.db.get(args.collectionId);
      if (!collection || collection.userId !== user._id) {
        return [];
      }
      const memberships = await ctx.db
        .query("libraryItemCollections")
        .withIndex("by_collection", (q) =>
          q.eq("collectionId", args.collectionId!)
        )
        .collect();
      collectionItemIds = new Set(memberships.map((m) => m.libraryItemId as string));
      if (collectionItemIds.size === 0) return [];
    }

    // Choose query path: search mode vs browse mode
    const hasSearch = args.searchText && args.searchText.trim().length > 0;

    let items;
    if (hasSearch) {
      // Search path: use search index, then collect all
      const q = ctx.db
        .query("libraryItems")
        .withSearchIndex("search_library", (q) =>
          q.search("searchableText", args.searchText!).eq("userId", user._id)
        );
      items = await q.collect();
    } else {
      // Browse path: use sort index, then collect all
      const indexMap = {
        savedAt: "by_user_savedAt",
        rating: "by_user_rating",
        publicationYear: "by_user_year",
      } as const;
      const q = ctx.db
        .query("libraryItems")
        .withIndex(indexMap[args.sortBy], (q) => q.eq("userId", user._id))
        .order(args.sortOrder);
      items = await q.collect();
    }

    // Apply filters (same logic as browseLibrary/searchLibrary)
    return items.filter((item) => {
      if (collectionItemIds && !collectionItemIds.has(item._id as string)) return false;
      if (args.minRating !== undefined && item.rating < args.minRating) return false;
      if (args.savedAfter !== undefined && item.savedAt < args.savedAfter) return false;
      if (args.yearFrom !== undefined && (item.publicationYear == null || item.publicationYear < args.yearFrom)) return false;
      if (args.yearTo !== undefined && (item.publicationYear == null || item.publicationYear > args.yearTo)) return false;
      return true;
    });
  },
});
```

Note: Uses `.collect()` which returns up to 1024 items -- acceptable per decision [23-01].

**2. Create `lib/library/export.ts`:**

Create a new file with library-specific export helpers that delegate to the existing `downloadAsFile` and `jsonToCSV` from `lib/utils.ts`.

```typescript
import { downloadAsFile, jsonToCSV } from "@/lib/utils";
import type { Doc } from "@/convex/_generated/dataModel";

interface LibraryExportItem {
  title: string;
  authors: string;
  doi: string;
  publicationYear: number | string;
  journal: string;
  rating: number;
  citedByCount: number | string;
  isOpenAccess: boolean | string;
  oaStatus: string;
  openalexId: string;
  savedAt: string;
  updatedAt: string;
  abstract: string;
}

function formatForExport(item: Doc<"libraryItems">): LibraryExportItem {
  return {
    title: item.title,
    authors: item.authors ?? "",
    doi: item.doi ?? "",
    publicationYear: item.publicationYear ?? "",
    journal: item.journal ?? "",
    rating: item.rating,
    citedByCount: item.citedByCount ?? "",
    isOpenAccess: item.isOpenAccess ?? "",
    oaStatus: item.oaStatus ?? "",
    openalexId: item.openalexId,
    savedAt: new Date(item.savedAt).toISOString(),
    updatedAt: new Date(item.updatedAt).toISOString(),
    abstract: item.abstract ?? "",
  };
}

export function exportLibraryAsCSV(items: Doc<"libraryItems">[]) {
  const rows = items.map(formatForExport);
  const csv = jsonToCSV(rows);
  const filename = `library-export-${new Date().toISOString().slice(0, 10)}.csv`;
  downloadAsFile(csv, filename, "csv");
}

export function exportLibraryAsJSON(items: Doc<"libraryItems">[]) {
  const data = {
    exportedAt: new Date().toISOString(),
    totalItems: items.length,
    items: items.map(formatForExport),
  };
  const filename = `library-export-${new Date().toISOString().slice(0, 10)}.json`;
  downloadAsFile(JSON.stringify(data, null, 2), filename, "json");
}
```

Key design choices:
- Strip `_id`, `_creationTime`, `userId`, `searchableText` from exported data (internal fields).
- Convert `savedAt`/`updatedAt` from Unix timestamps to ISO date strings for universal parseability.
- Use empty string `""` for optional fields that are null/undefined (cleaner CSV output than "null").
- Date-stamped filenames (`library-export-2026-03-01.csv`) instead of epoch timestamps for human readability.
