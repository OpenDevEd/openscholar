**File 1: Create `convex/lib/extractionLimit.ts`**

Create a new file with a single exported async function that resolves a user's extraction limit from their subscription plan. Follow the `convex/subscriptions/queries.ts:getTier` pattern exactly:

```typescript
import type { QueryCtx, MutationCtx } from "../_generated/server";
import type { Id } from "../_generated/dataModel";
import { EXTRACTION_LIMITS_BY_TIER } from "./queryBounds";
import { resolveEffectivePlan } from "../../lib/billing/entitlements";
import { selectCanonicalSubscription } from "../subscriptions/lib/selectCanonicalSubscription";

/**
 * Resolve the extraction paper limit for a specific user based on their subscription plan.
 *
 * Uses the same subscription resolution chain as getTier:
 * 1. Fetch all subscriptions for the user (cardinality: ~1-5)
 * 2. Select canonical subscription via priority logic
 * 3. Resolve effective plan (handles canceled, trialing, past_due)
 * 4. Look up plan's limit in EXTRACTION_LIMITS_BY_TIER
 *
 * Returns the per-job paper limit for the user's effective plan.
 */
export async function getExtractionLimitForUser(
  ctx: QueryCtx | MutationCtx,
  userId: Id<"users">,
): Promise<number> {
  const subscriptions = await ctx.db
    .query("subscriptions")
    .withIndex("by_user", (q) => q.eq("userId", userId))
    .order("desc")
    .collect();

  const subscription = selectCanonicalSubscription(subscriptions);
  const plan = resolveEffectivePlan(subscription);
  return EXTRACTION_LIMITS_BY_TIER[plan];
}
```

Key implementation notes:
- Context type is `QueryCtx | MutationCtx` (not ActionCtx) because this helper reads from the DB.
- Uses `.collect()` not `.first()` because `selectCanonicalSubscription` needs the full list to apply priority logic (Pitfall 2 from RESEARCH.md).
- The import from `"../../lib/billing/entitlements"` is a cross-boundary import but safe because `entitlements.ts` is pure TypeScript with no Node.js builtins.

```typescript
/**
 * Reactive query exposing the authenticated user's per-job extraction limit.
 *
 * Returns { limit, plan } where limit is the maximum papers per extraction job
 * for the user's current subscription plan, and plan is the resolved plan name.
 * Unauthenticated users receive { limit: 0, plan: "free" }.
 *
 * Phase 102 consumes this for the "X / 1000" counter display.
 */
export const getExtractionLimit = query({
  args: {},
  returns: v.object({ limit: v.number(), plan: v.string() }),
  handler: async (ctx) => {
    const user = await getCurrentUser(ctx);
    if (!user) return { limit: 0, plan: "free" };

    const limit = await getExtractionLimitForUser(ctx, user._id);

    // Re-derive plan for client display. The subscription lookup is cheap
    // (cardinality ~1-5 rows) and already cached in the reactive query.
    const subscriptions = await ctx.db
      .query("subscriptions")
      .withIndex("by_user", (q) => q.eq("userId", user._id))
      .order("desc")
      .collect();
    const subscription = selectCanonicalSubscription(subscriptions);
    const plan = resolveEffectivePlan(subscription);

    return { limit, plan };
  },
});
```
