```typescript
import { resolveEffectivePlan } from "../../lib/billing/entitlements";
import { getPlanTier } from "../lib/constants";
import { createError } from "../lib/errors";
import { selectCanonicalSubscription } from "../subscriptions/lib/selectCanonicalSubscription";

export async function requirePlanTier(
  ctx: MutationCtx,
  userId: Id<"users">,
  minTier: number,
  featureName: string
): Promise<void> {
  const subscriptions = await ctx.db
    .query("subscriptions")
    .withIndex("by_user", (q) => q.eq("userId", userId))
    .order("desc")
    .collect();

  const canonical = selectCanonicalSubscription(subscriptions);
  const plan = resolveEffectivePlan(canonical);
  const tier = getPlanTier(plan);

  if (tier < minTier) {
    throw createError(
      "EXTRACTION_PLAN_REQUIRED",
      `${featureName} requires a higher plan. Please upgrade to continue.`
    );
  }
}
```

```typescript
export function resolveEffectivePlan(subscription: SubscriptionLike): SubscriptionPlan {
  if (!subscription) return "free";
  return isEntitledStatus(subscription.status) ? subscription.plan : "free";
}
// Entitled statuses: "active", "trialing", "past_due"
// Non-entitled: "canceled", "incomplete", "incomplete_expired", "unpaid", "paused"
```

### describe('selectCanonicalSubscription') -- PLAN-05

Test the 6-level priority ordering. Each test creates a subscription array with entries designed to verify one priority level is selected over lower-priority alternatives.

1. **Returns null for empty array** -- `expect(selectCanonicalSubscription([])).toBeNull()`.

2. **Priority 1: non-canceled paid with Stripe ID wins over all** -- Create array with: active free sub, active go sub with stripeSubscriptionId. Assert the go sub with Stripe ID is returned.
```typescript
const subs = [
  makeSub({ _id: "free_active", plan: "free", status: "active" }),
  makeSub({ _id: "go_stripe", plan: "go", status: "active", stripeSubscriptionId: "sub_123" }),
];
expect(selectCanonicalSubscription(subs)).toBe(subs[1]);
```

3. **Priority 2: non-canceled paid without Stripe ID wins over free** -- Create array with: active free sub, active pro sub WITHOUT stripeSubscriptionId. No sub has Stripe ID. Assert pro sub is returned.
```typescript
const subs = [
  makeSub({ _id: "free_active", plan: "free", status: "active" }),
  makeSub({ _id: "pro_no_stripe", plan: "pro", status: "active" }),
];
expect(selectCanonicalSubscription(subs)).toBe(subs[1]);
```

4. **Priority 3: non-canceled free wins over canceled free** -- Create array with: canceled free sub, active free sub. Assert active free is returned.
```typescript
const subs = [
  makeSub({ _id: "free_canceled", plan: "free", status: "canceled" }),
  makeSub({ _id: "free_active", plan: "free", status: "active" }),
];
expect(selectCanonicalSubscription(subs)).toBe(subs[1]);
```

5. **Priority 4: any free wins over canceled paid with Stripe ID** -- Create array with: canceled go sub with stripeSubscriptionId, canceled free sub. Assert canceled free sub is returned (priority 4 > priority 5).
```typescript
const subs = [
  makeSub({ _id: "go_canceled_stripe", plan: "go", status: "canceled", stripeSubscriptionId: "sub_456" }),
  makeSub({ _id: "free_canceled", plan: "free", status: "canceled" }),
];
expect(selectCanonicalSubscription(subs)).toBe(subs[1]);
```

6. **Priority 5: any with Stripe ID wins over record without** -- Create array with: canceled go sub WITHOUT stripeSubscriptionId, canceled pro sub WITH stripeSubscriptionId. Neither is free. Assert the one with Stripe ID is returned.

7. **Priority 6: falls back to first record** -- Create array with: two canceled non-free subs, neither has stripeSubscriptionId. Assert first record is returned.
