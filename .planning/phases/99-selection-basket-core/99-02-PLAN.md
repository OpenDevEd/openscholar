    4. Component body:
       - Hydration gate: `const hasHydrated = useExtractionBasketStore((s) => s._hasHydrated);`
       - Paper data: `const papers = useExtractionBasketStore((s) => s.papers);`
       - Count: `const count = useExtractionBasketStore((s) => s.papers.size);`
       - Remove action: `const remove = useExtractionBasketStore((s) => s.remove);`
       - Router: `const router = useRouter();`
       - Translations: `const t = useTranslations("extraction.basket");`

       - `entriesBySource` computed via `useMemo`:
         ```typescript
         const entriesBySource = useMemo(() => {
           const grouped = new Map<BasketSource, BasketEntry[]>();
           for (const entry of papers.values()) {
             const existing = grouped.get(entry.source) ?? [];
             existing.push(entry);
             grouped.set(entry.source, existing);
           }
           return grouped;
         }, [papers]);
         ```

       - `handleRemove` callback: `useCallback((id: string) => remove(id), [remove])`

       - `handleCTA` callback: `useCallback(() => { router.push("/extraction/new"); onClose(); }, [router, onClose])`

       - Accordion `defaultValue`: all source keys that have entries, so all sections start expanded:
         ```typescript
         const expandedSources = useMemo(
           () => Array.from(entriesBySource.keys()),
           [entriesBySource]
         );
         ```

    5. JSX structure (per UI-SPEC Drawer Layout):
       ```
       <Sheet open={isOpen} onOpenChange={onClose}>
         <SheetContent side="right" className="flex flex-col p-0">
           <SheetHeader className="p-4 border-b">
             <SheetTitle>{t("title")}</SheetTitle>
             <SheetDescription>
               <span className="text-primary font-semibold">{count}</span>{" "}
               {t("description", { count })}
             </SheetDescription>
           </SheetHeader>

           <ScrollArea className="flex-1">
             {!hasHydrated ? null : count === 0 ? (
               <BasketEmptyState />
             ) : (
               <div className="p-4">
                 <Accordion type="multiple" defaultValue={expandedSources}>
                   {Array.from(entriesBySource.entries()).map(([source, entries]) => (
                     <BasketSourceSection
                       key={source}
                       source={source}
                       entries={entries}
                       onRemove={handleRemove}
                     />
                   ))}
                 </Accordion>

           </ScrollArea>

               disabled={count === 0}

             >
               {t("cta")}
             </Button>
           </SheetFooter>
         </SheetContent>
       </Sheet>
       ```
