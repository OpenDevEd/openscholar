export function createSidebarStore(persistKey: string) {
  return create<FilterSidebarState>()(
    persist(
      (set) => ({
        isCollapsed: false,
        _hasHydrated: false,
        setCollapsed: (collapsed) => set({ isCollapsed: collapsed }),
        toggle: () => set((state) => ({ isCollapsed: !state.isCollapsed })),
        setHasHydrated: (state) => set({ _hasHydrated: state }),
      }),
      {
        name: persistKey,
        version: 1,
        migrate: (persistedState: unknown, version: number) => {
          if (version < 1) return { isCollapsed: false };
          return persistedState;
        },
        storage: createJSONStorage(() => localStorage),
        partialize: (state) => ({ isCollapsed: state.isCollapsed }),
        onRehydrateStorage: () => (state) => {
          state?.setHasHydrated(true);
        },
      }
    )
  );
}

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| User input -> normalizeOpenAlexId | Untrusted string input from paste, URL, or manual entry crosses into normalization |
| sessionStorage -> store rehydration | Stored data could be tampered with in browser DevTools |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-99-01 | Tampering | sessionStorage basket data | accept | Client-side-only staging data with no auth implications; user can only tamper with their own basket. Job creation in Phase 100/103 validates server-side via createExtractionJob. |
| T-99-02 | Spoofing | normalizeOpenAlexId input | mitigate | Regex strictly matches `W\d+` pattern only; rejects all non-OpenAlex formats (DOIs, PMIDs, arbitrary strings). Null return on invalid input prevents injection of non-work IDs into the basket. |
| T-99-03 | Denial of Service | sessionStorage quota exhaustion | accept | 1000 entries x ~300 bytes = ~300KB, well under the 5MB sessionStorage limit. Cap enforcement happens in Phase 102 (LIM-05). |
| T-99-04 | Information Disclosure | Paper titles/DOIs in sessionStorage | accept | sessionStorage is same-origin, tab-scoped, and clears on tab close. No PII beyond paper metadata the user already has access to. |
</threat_model>
