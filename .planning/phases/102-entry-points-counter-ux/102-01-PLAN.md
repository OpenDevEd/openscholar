```typescript
"use client";

import { useExtractionBasketStore } from "@/lib/store/extraction-basket";
import { useProtectedQuery } from "@/lib/auth/use-auth-query";
import { api } from "@/convex/_generated/api";
import { Progress } from "@/components/ui/progress";
import { useTranslations } from "next-intl";
import { cn } from "@/lib/utils";

export function ExtractionCounter({ className }: { className?: string }) {
  const t = useTranslations("extraction.counter");
  const count = useExtractionBasketStore((s) => s.papers.size);
  const limitData = useProtectedQuery(api.extraction.queries.getExtractionLimit, {});
  const limit = limitData?.limit ?? 1000;
  const ratio = limit > 0 ? count / limit : 0;

  return (
    <div className={cn("space-y-1.5", className)}>
      <div className="flex items-center justify-between text-sm">
        <span className="font-medium">{t("label", { count, limit })}</span>
        {ratio >= 1 && (
          <span className="text-xs text-destructive font-medium">
            {t("limitReached")}
          </span>
        )}
        {ratio >= 0.8 && ratio < 1 && (
          <span className="text-xs text-amber-600 dark:text-amber-400 font-medium">
            {t("limitWarning")}
          </span>
        )}
      </div>
      <Progress
        value={Math.min(ratio * 100, 100)}
        className={cn(
          "h-2",
          ratio >= 1 && "[&>[data-slot=progress-indicator]]:bg-destructive",
          ratio >= 0.8 && ratio < 1 && "[&>[data-slot=progress-indicator]]:bg-amber-500 dark:[&>[data-slot=progress-indicator]]:bg-amber-400",
        )}
      />
    </div>
  );
}
```

Key requirements from the UI-SPEC:
- 3 visual states: Default (ratio < 0.8, inherits bg-primary), Warning (0.8 <= ratio < 1.0, amber indicator + amber text), Full (ratio >= 1.0, destructive indicator + destructive text)
- Ratio computed from `count / limit` per D-02 -- NEVER hardcoded against 1000
- `limit` fallback is `?? 1000` while query loads
- Progress value clamped to `Math.min(ratio * 100, 100)`
- Component is self-contained (reads own data via Zustand selector and Convex query), accepts only optional `className` prop
- Uses `useProtectedQuery` (not bare `useQuery`) to avoid flash during Clerk auth transitions
- Target the indicator fill with `[&>[data-slot=progress-indicator]]` descendant selector (per RESEARCH.md Pitfall 1)
- Uses `useTranslations("extraction.counter")` for all text

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-102-01 | Tampering | Client-side cap bypass (user modifies JS to skip truncation in useAddToBasket) | accept | Backend `createExtractionJob` mutation hard-rejects over-limit submissions (Phase 100 LIM-07). Client cap is UX convenience, not security boundary. |

</threat_model>
