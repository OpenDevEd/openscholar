```typescript
handler: async (ctx, args) => {
  const subscriptions = await ctx.db
    .query("subscriptions")
    .withIndex("by_stripe_sub", (q) => q.eq("stripeSubscriptionId", args.stripeSubscriptionId))
    .order("desc")
    .collect();
  const subscription = selectCanonicalSubscription(subscriptions);

  if (!subscription) return null;

  const now = Date.now();
  const patch: Record<string, any> = {
    status: "past_due" satisfies SubscriptionStatus,
    updatedAt: now,
  };

  // Set pastDueSince only on the FIRST past_due transition
  // (not overwritten by subsequent payment failure events)
  if (!subscription.pastDueSince) {
    patch.pastDueSince = now;
  }

  await ctx.db.patch(subscription._id, patch);

  return null;
},
```

**Step 5: Enhance `checkSubscriptionStatus` cron (`convex/maintenance/internal.ts`)**

Replace the existing `checkSubscriptionStatus` handler with logic that:
1. Finds subscriptions with `past_due` status
2. Checks if `pastDueSince` + 7 days < now (grace period expired)
3. Auto-downgrades expired subscriptions to free plan

```typescript
export const checkSubscriptionStatus = internalMutation({
  args: {},
  returns: v.null(),
  handler: async (ctx) => {
    const now = Date.now();
    const GRACE_PERIOD_MS = 7 * 24 * 60 * 60 * 1000; // 7 days per locked decision

    // Find past_due subscriptions
    const pastDueSubs = await ctx.db
      .query("subscriptions")
      .withIndex("by_status", (q) => q.eq("status", "past_due"))
      .collect();

    let downgradeCount = 0;

    for (const sub of pastDueSubs) {
      const pastDueSince = sub.pastDueSince ?? sub.updatedAt;
      if (now - pastDueSince > GRACE_PERIOD_MS) {
        // Grace period expired -- downgrade to free plan
        // Per locked decision: automatically downgrade to free plan after grace period
        await ctx.db.patch(sub._id, {
          status: "canceled",
          plan: "free",
          canceledAt: now,
          updatedAt: now,
          creditsPerMonth: 0,
          pastDueSince: undefined,
          // stripeSubscriptionId preserved for audit trail (consistent with SUB-03 fix)
        });
        downgradeCount++;
        console.warn(`[checkSubscriptionStatus] Grace period expired, downgraded subscription ${sub._id} to free`, {
          userId: sub.userId,
          pastDueSince,
          gracePeriodDays: 7,
        });
      }
    }

    // Also check for active subscriptions past their period end (existing behavior, kept)
    const activeSubscriptions = await ctx.db
      .query("subscriptions")
      .withIndex("by_status", (q) => q.eq("status", "active"))
      .collect();

    for (const sub of activeSubscriptions) {
      if (sub.currentPeriodEnd && sub.currentPeriodEnd < now) {
        console.warn(`[checkSubscriptionStatus] Subscription ${sub._id} past period end, needs attention`);
      }
    }

    if (downgradeCount > 0) {
      console.log(`[checkSubscriptionStatus] Downgraded ${downgradeCount} past-due subscriptions after grace period`);
    }

    return null;
  },
});
```
