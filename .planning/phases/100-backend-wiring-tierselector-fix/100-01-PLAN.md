      name: "google/gemini-3-flash-preview",       // Index 1: middle

        provider: { allowFallbacks: true },
      }),
    },
    {
      name: "anthropic/claude-haiku-4.5",          // Index 2: most capable
      model: openrouter("anthropic/claude-haiku-4.5", {
        plugins: [{ id: "response-healing" }],
        provider: { allowFallbacks: true },
      }),
    },
  ];

  switch (tier) {
    case "cheap":

    case undefined:

}
