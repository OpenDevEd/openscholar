```typescript
async function handleExport(
  jobId: Id<"extractionJobs">,
  fmt: "csv" | "json" | "xlsx"
) {
  setExportingJobId(jobId);
  try {
    const data = await convex.query(
      api.extraction.queries.getExtractionResultsForExport,
      { jobId }
    );
    if (!data) return;
    switch (fmt) {
      case "csv":  exportExtractionToCSV(data); break;
      case "json": exportExtractionToJSON(data); break;
      case "xlsx": await exportExtractionToXLSX(data); break;
    }
  } catch (err) {
    console.error("Export failed:", err);
    showStyledToast("error", t("export.incompleteWarning"));
  } finally {
    setExportingJobId(null);
  }
}
```

```typescript
function ExtractionFooterButton({
  onClick,
  isCollapsed,
}: {
  onClick: () => void;
  isCollapsed: boolean;
}) {
  const t = useTranslations("extraction");
  const { data: jobs } = useProtectedQuery(
    api.extraction.queries.listUserExtractionJobs,
    { limit: 1 }
  );

  const activeJob = jobs?.[0];
  const isActive = activeJob && (activeJob.status === "pending" || activeJob.status === "processing");

  const { data: progress } = useProtectedQuery(
    api.extraction.queries.getExtractionProgress,
    isActive ? { jobId: activeJob._id } : "skip"
  );

  const progressPercent = progress?.percentComplete ?? null;
  const circumference = 2 * Math.PI * 10;

  const icon = (
    <span className="relative inline-flex items-center justify-center">
      <FlaskConical className="h-[1.1rem] w-[1.1rem]" />
      {progressPercent !== null && (
        <svg viewBox="0 0 24 24" className="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 scale-[2] origin-center" fill="none">
          <circle cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="1.5" opacity="0.1" />
          <circle cx="12" cy="12" r="10"
            stroke="oklch(0.70 0.17 54)"
            strokeWidth="1.5"
            strokeDasharray={circumference}
            strokeDashoffset={circumference - (circumference * progressPercent) / 100}
            strokeLinecap="round"
            transform="rotate(-90 12 12)"
            className="transition-[stroke-dashoffset] duration-500 ease-out"
          />
        </svg>
      )}
    </span>
  );

  // Collapsed/expanded variants -- follow ScreeningFooterButton exactly
  // Collapsed: Tooltip-wrapped icon button
  // Expanded: Full-width button with icon + label + percentage
}
```
