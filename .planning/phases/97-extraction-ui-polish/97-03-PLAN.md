```typescript
const convex = useConvex();
const [isExporting, setIsExporting] = useState(false);

async function handleExport(fmt: "csv" | "json" | "xlsx") {
  setIsExporting(true);
  try {
    const data = await convex.query(
      api.extraction.queries.getExtractionResultsForExport,
      { jobId: jobId as Id<"extractionJobs"> }
    );
    if (!data) return;
    switch (fmt) {
      case "csv":  exportExtractionToCSV(data); break;
      case "json": exportExtractionToJSON(data); break;
      case "xlsx": await exportExtractionToXLSX(data); break;
    }
  } catch (err) {
    console.error("Export failed:", err);
    showStyledToast("error", t("export.incompleteWarning"));
  } finally {
    setIsExporting(false);
  }
}
```
