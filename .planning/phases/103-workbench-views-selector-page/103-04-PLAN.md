```typescript
"use client";

import { useRef, useEffect } from "react";
import { useVirtualizer } from "@tanstack/react-virtual";
import { useTranslations } from "next-intl";
import type { Table as TanStackTable } from "@tanstack/react-table";
import { PaperResultCard } from "@/components/extraction/paper-result-card";
import type {
  WorkbenchPaper,
  ExtractionQuestion,
} from "@/lib/extraction/workbench-columns";
import type { EvidenceDrawerProps } from "@/components/extraction/workbench-shell";

interface CardViewProps {
  table: TanStackTable<WorkbenchPaper>;
  questions: ExtractionQuestion[];
  onEvidenceClick: (props: EvidenceDrawerProps) => void;
}

export function CardView({ table, questions, onEvidenceClick }: CardViewProps) {
  const t = useTranslations("extraction.workbench");
  const parentRef = useRef<HTMLDivElement>(null);

  // Get paginated rows from TanStack Table (D-17: pagination controls how many the virtualizer receives)
  const rows = table.getRowModel().rows;

  const virtualizer = useVirtualizer({
    count: rows.length,
    getScrollElement: () => parentRef.current,
    estimateSize: () => 280, // D-02: ~280px estimated card height
    overscan: 5,             // D-02: 5 rows above and below viewport
  });

  // Pitfall 2: reset scroll position when data changes (filter/sort/pagination)
  useEffect(() => {
    virtualizer.scrollToIndex(0);
  }, [rows.length, virtualizer]);

  if (rows.length === 0) {
    return (
      <div className="flex min-h-[30vh] items-center justify-center">
        <p className="text-sm text-muted-foreground">{t("splitEmptyHeading")}</p>
      </div>
    );
  }

  return (
    <div
      ref={parentRef}
      className="h-[calc(100vh-280px)] overflow-y-auto"
    >
      <div
        style={{
          height: virtualizer.getTotalSize(),
          width: "100%",
          position: "relative",
        }}
      >
        {virtualizer.getVirtualItems().map((virtualRow) => {
          const paper = rows[virtualRow.index].original;
          return (
            <div
              key={virtualRow.key}
              data-index={virtualRow.index}
              ref={virtualizer.measureElement}
              style={{
                position: "absolute",
                top: 0,
                left: 0,
                width: "100%",
                transform: `translateY(${virtualRow.start}px)`,
              }}
            >
              <div className="pb-4">
                <PaperResultCard
                  paper={paper}
                  questions={questions}
                  index={virtualRow.index}
                />
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
}
```

```typescript
      role="row"

          {paper.parsingSource}

      )}

    </div>
  );
```

```typescript
"use client";

import { useState, useRef, useCallback, useMemo } from "react";
import { useTranslations } from "next-intl";
import type { Table as TanStackTable } from "@tanstack/react-table";
import { PaperResultCard } from "@/components/extraction/paper-result-card";
import { SplitViewRow } from "@/components/extraction/split-view-row";
import { ChevronLeft, ChevronRight, FileText } from "lucide-react";
import { Button } from "@/components/ui/button";
import { ScrollArea } from "@/components/ui/scroll-area";
import type {
  WorkbenchPaper,
  ExtractionQuestion,
} from "@/lib/extraction/workbench-columns";
import type { EvidenceDrawerProps } from "@/components/extraction/workbench-shell";
import { cn } from "@/lib/utils";

interface SplitViewProps {
  table: TanStackTable<WorkbenchPaper>;
  questions: ExtractionQuestion[];
  onEvidenceClick: (props: EvidenceDrawerProps) => void;
}

const SPLIT_PAGE_SIZE = 10; // D-05: denser pagination for left panel

export function SplitView({ table, questions, onEvidenceClick }: SplitViewProps) {
  const t = useTranslations("extraction.workbench");
  const [selectedPaperId, setSelectedPaperId] = useState<string | null>(null);
  const [currentPage, setCurrentPage] = useState(0);
  const detailRef = useRef<HTMLDivElement>(null);

  // Get all filtered/sorted rows from the table
  const allRows = table.getFilteredRowModel().rows;
  const totalPages = Math.max(1, Math.ceil(allRows.length / SPLIT_PAGE_SIZE));
  const pageRows = useMemo(
    () => allRows.slice(currentPage * SPLIT_PAGE_SIZE, (currentPage + 1) * SPLIT_PAGE_SIZE),
    [allRows, currentPage],
  );

  // Find the selected paper from all rows (not just current page)
  const selectedPaper = useMemo(
    () => allRows.find((row) => row.original._id === selectedPaperId)?.original ?? null,
    [allRows, selectedPaperId],
  );

  const handleSelect = useCallback((id: string) => {
    setSelectedPaperId(id);
    // Scroll right panel to top on paper change (D-05)
    detailRef.current?.scrollTo({ top: 0, behavior: "smooth" });
  }, []);

  const rangeStart = allRows.length === 0 ? 0 : currentPage * SPLIT_PAGE_SIZE + 1;
  const rangeEnd = Math.min((currentPage + 1) * SPLIT_PAGE_SIZE, allRows.length);

  return (
    <div className="flex h-[calc(100vh-280px)] min-h-[400px]">
      {/* Left panel: 40% width, paginated list */}
      <div className="w-[40%] border-r border-border flex flex-col">
        <div className="flex-1 overflow-y-auto" role="grid" aria-label="Paper list">
          {pageRows.map((row) => (
            <SplitViewRow
              key={row.original._id}
              paper={row.original}
              isSelected={row.original._id === selectedPaperId}
              onSelect={handleSelect}
            />
          ))}
        </div>
        {/* Left panel pagination */}
        {totalPages > 1 && (
          <div className="flex items-center justify-between border-t border-border px-3 py-2">
            <span className="text-xs text-muted-foreground">
              {rangeStart}-{rangeEnd} / {allRows.length}
            </span>
            <div className="flex items-center gap-1">
              <Button
                type="button"
                variant="ghost"
                size="icon"
                className="h-7 w-7"
                onClick={() => setCurrentPage((p) => Math.max(0, p - 1))}
                disabled={currentPage === 0}
              >
                <ChevronLeft className="h-3.5 w-3.5" />
              </Button>
              <Button
                type="button"
                variant="ghost"
                size="icon"
                className="h-7 w-7"
                onClick={() => setCurrentPage((p) => Math.min(totalPages - 1, p + 1))}
                disabled={currentPage >= totalPages - 1}
              >
                <ChevronRight className="h-3.5 w-3.5" />
              </Button>
            </div>
          </div>
        )}
      </div>

      {/* Right panel: 60% width, detail */}
      <div ref={detailRef} className="w-[60%] overflow-y-auto p-4">
        {selectedPaper ? (
          <PaperResultCard
            paper={selectedPaper}
            questions={questions}
            index={0}
          />
        ) : (
          <div className="flex h-full flex-col items-center justify-center text-center">
            <FileText className="h-12 w-12 text-muted-foreground/30 mb-4" />
            <h3 className="text-base font-semibold text-muted-foreground">
              {t("splitEmptyHeading")}
            </h3>
            <p className="text-sm text-muted-foreground mt-1 max-w-xs">
              {t("splitEmptyBody")}
            </p>
          </div>
        )}
      </div>
    </div>
  );
}
```

Key points:
- Fixed 40%/60% split, NOT resizable (D-06)
- Left panel paginated at page size 10 (D-05)
- `selectedPaperId` is local state, `useState<string | null>(null)` (D-04)
- Right panel scrolls to top on paper change (D-05)
- Selected row has `bg-accent` + `border-l-2 border-l-primary` (UI-SPEC)
- Empty state in right panel when no paper selected

NOTE: Mobile fallback (D-06: < 768px -> table view + toast) will be wired in Plan 05 when WorkbenchShell integrates all views together. The shell reads `useIsTablet` or a media query and auto-switches activeView.
