**A) Create `components/extraction/pagination-controls.tsx`**

Per D-18 and the library-page-client.tsx pagination pattern (lines 564-611):

```typescript
"use client";

import { ChevronLeft, ChevronRight } from "lucide-react";
import { useTranslations } from "next-intl";
import type { Table as TanStackTable } from "@tanstack/react-table";
import { Button } from "@/components/ui/button";
import type { WorkbenchPaper } from "@/lib/extraction/workbench-columns";

interface PaginationControlsProps {
  table: TanStackTable<WorkbenchPaper>;
}

export function PaginationControls({ table }: PaginationControlsProps) {
  const t = useTranslations("extraction.workbench");
  const { pageIndex, pageSize } = table.getState().pagination;
  const filteredTotal = table.getFilteredRowModel().rows.length;
  const rangeStart = filteredTotal === 0 ? 0 : pageIndex * pageSize + 1;
  const rangeEnd = Math.min((pageIndex + 1) * pageSize, filteredTotal);
  const totalPages = table.getPageCount();

  return (
    <div
      className="relative flex flex-col gap-4 p-4 sm:flex-row sm:items-center sm:justify-between"
      aria-live="polite"
    >
      <p className="text-sm font-semibold text-muted-foreground">
        {t("paginationShowing", { start: rangeStart, end: rangeEnd, total: filteredTotal })}
      </p>
      {totalPages > 1 && (
        <div className="flex items-center gap-2">
          <Button
            type="button"
            variant="outline"
            size="sm"
            onClick={() => table.previousPage()}
            disabled={!table.getCanPreviousPage()}
            className="rounded-xl h-11"
          >
            <ChevronLeft className="mr-1 h-4 w-4" />
            {t("paginationPrevious")}
          </Button>
          <span className="min-w-[110px] text-center text-sm text-muted-foreground">
            {pageIndex + 1} / {totalPages}
          </span>
          <Button
            type="button"
            variant="outline"
            size="sm"
            onClick={() => table.nextPage()}
            disabled={!table.getCanNextPage()}
            className="rounded-xl h-11"
          >
            {t("paginationNext")}
            <ChevronRight className="ml-1 h-4 w-4" />
          </Button>
        </div>
      )}
    </div>
  );
}
```

const handleExport = useCallback(async (fmt: "csv" | "json" | "xlsx") => {
  if (!jobId) return;
  setIsExporting(true);
  try {
    const exportData = await convex.query(
      api.extraction.queries.getExtractionResultsForExport,
      {
        jobId: jobId as Id<"extractionJobs">,
        ...(hasActiveFilters ? { paperIds: filteredPaperIds as Id<"extractionPapers">[] } : {}),
      },
    );
    if (!exportData) return;
    switch (fmt) {
      case "csv": exportExtractionToCSV(exportData as ExtractionExportData); break;
      case "json": exportExtractionToJSON(exportData as ExtractionExportData); break;
      case "xlsx": await exportExtractionToXLSX(exportData as ExtractionExportData); break;
    }
  } catch (err) {
    console.error("Export failed:", err);
    showStyledToast("error", t("exportButton"));
  } finally {
    setIsExporting(false);
  }
}, [jobId, convex, hasActiveFilters, filteredPaperIds, t]);
